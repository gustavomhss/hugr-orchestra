const SESSION = "ses_f1511c48bffeDAL863Z2Lw4p1b"
const db = new Database(B + "/trace.db")
const rows = db.query("select id, data from message where session_id = ? order by time_created, id").all(SESSION) as any[]
const messages = rows.map((row) => ({ info: { ...JSON.parse(row.data), id: row.id, sessionID: SESSION },
  parts: (db.query("select id, message_id, data from part where message_id = ? order by id").all(row.id) as any[]).map((p) => ({ ...JSON.parse(p.data), id: p.id, sessionID: SESSION, messageID: p.message_id })) })) as any[]
const agents = new Map((db.query("select id, agent from session where parent_id = ?").all(SESSION) as any[]).map((r) => [r.id, r.agent]))
const fold = (s: string) => s.toLowerCase().normalize("NFD").replace(/\p{M}/gu, "").replace(/\s+/g, " ").trim()
const history = messages.slice(0, 37)
const captured = snapshot(SESSION as any, history, undefined, true, 32000)!
const delegations = Object.fromEntries([...new Set(history.flatMap((m: any) => m.parts.flatMap((p: any) => child(p) ?? [])))].map((id) => [id, { member: agents.get(id), status: undefined }]))
const host = { history, delegations, member: false } as any
const ctx = scope(captured, host)
const text = readFileSync(B + "/seed-3/evidence/v4/pass1.reply1.txt", "utf8").trim()
const full = decode({ text, snapshot: captured, producerID: "ses_diag" as any, host, ceiling: 14039 }) as any
console.log("full decode dropped:", full.dropped, "items:", full.artifact?.items.length)
for (const [index, op] of JSON.parse(text).ops.entries()) {
  const single = { ...op, fields: { ...op.fields } }; delete single.key; if (single.fields.needs) single.fields.needs = []
  const one = decode({ text: JSON.stringify({ ops: [single] }), snapshot: captured, producerID: "ses_diag" as any, host, ceiling: 14039 }) as any
  if (!one.dropped) continue
  const name = "error" in op.fields ? "error" : "value"
  const needle = fold(op.fields[name])
  const where: string[] = []
  for (const source of ctx.covered as any[]) {
    const st = source.part?.type === "tool" ? source.part.state : undefined
    if (source.text && fold(source.text).includes(needle)) where.push(source.alias + " text")
    for (const [k, v] of Object.entries(st?.input ?? {})) if (fold(typeof v === "string" ? v : JSON.stringify(v)).includes(needle)) where.push(`${source.alias} input.${k}${KEY_ARGS.includes(k) ? " (KEY_ARG)" : " (not KEY_ARG)"}`)
    const out = st?.output ?? st?.error
    if (out && fold(out).includes(needle)) where.push(source.alias + " output/error")
    if (st?.metadata && fold(JSON.stringify(st.metadata)).includes(needle)) where.push(source.alias + " metadata")
  }
  const req = fold(readFileSync(B + "/seed-3/exchange/v4-pass1.request.md", "utf8"))
  console.log(`op ${index + 1} ${op.section} key=${op.key} src=${op.src} ${name}=${JSON.stringify(op.fields[name])}\n   name=${op.fields.name ?? op.fields.tried}\n   in request: ${req.includes(needle)}; occurs in: ${where.join("; ") || "no covered source"}`)
}
