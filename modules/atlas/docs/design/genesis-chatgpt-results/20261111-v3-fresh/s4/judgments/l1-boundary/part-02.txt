export async function ask(revisionId: string, text: string): Promise<QuestionResult> {
  const store = requireStore()
  const revision = store.revision(revisionId)
  if (!revision) throw new Error("rejected")
  if (revision.fill === "failed" || store.failed(revision.locationId)) throw new Error("rejected")
  const stale = store.chunks().some((chunk) => {
    const source = store.sources().find((row) => row.id === chunk.sourceId)
    if (!source || source.locationId !== revision.locationId) return false
    return source.contentHash !== chunk.sourceHash
  })
  if (stale) throw new Error("rejected")
  const boundary = revision.boundary ?? "document"
  const k = revision.k ?? store.pipeline(revision.locationId).k
  const matches = (await searchWords(revisionId, text)).filter((chunk) => store.chunkCut(chunk.id) === boundary)
  const owners = store
    .sources()
    .filter((row) => row.locationId === revision.locationId && row.text.includes(text))
    .sort((left, right) => left.path.localeCompare(right.path))
  const cut = block(matches, k)
  const found = diagnose({ owners, matches, kept: cut.kept, sentence: text })
  record(revisionId, found.span ? matches.map((chunk) => chunk.id) : matches.map((chunk) => chunk.id), cut.dropped.map((chunk) => chunk.id), k)

export async function retrievalProof(questionId: string, revisionId: string): Promise<boolean> {
  const store = requireStore()
  const question = store.questions().find((row) => row.id === questionId)
  if (!question) return false
  if (question.revisionId !== revisionId) return false
  const chunk = store
    .chunks()
    .find((item) => item.sourceId === question.span.sourceId && item.start === question.span.start && item.end === question.span.end)
  if (!chunk) return false
  const trace = store.trace(questionId)
  if (!trace) return false
  const revision = store.revision(revisionId)
  if (trace.k !== undefined && revision?.k !== undefined && trace.k !== revision.k) throw new Error("rejected")
  return promptChunkIds(trace).includes(chunk.id)

export function openRevision(locationId: string): Promise<StoredRevision> {
  const store = requireStore()
  const pipeline = store.pipeline(locationId)
  const members = store
    .sources()
    .filter((row) => row.locationId === locationId)
    .map((row) => ({ sourceId: row.id, contentHash: row.contentHash }))
    .sort((left, right) => (left.sourceId < right.sourceId ? -1 : left.sourceId > right.sourceId ? 1 : 0))
  const revision: StoredRevision = {
    id: store.allocate("rev"),
    snapshot: digest(members.map((row) => `${row.sourceId}\t${row.contentHash}`).join("\n")),
    pipelineHash: pipelineHash(pipeline),
    fill: fillFor(locationId, members),
    locationId,
    members,
    parser: pipeline.parser,
    boundary: pipeline.boundary,
    tokenizer: pipeline.tokenizer,
    k: pipeline.k,
  }
  store.putRevision(revision)
  return Promise.resolve(revision)

export async function extract(file: string): Promise<ReadBytes> {
  const blob = Bun.file(file)
  if (!(await blob.exists())) return { failed: "missing" }
  if (!file.toLowerCase().endsWith(".pdf")) return { text: await blob.text() }
  const bytes = new Uint8Array(await blob.arrayBuffer())
  const loaded = await import("unpdf")
  const pdf = await loaded.getDocumentProxy(bytes)
  const extracted = await loaded.extractText(pdf, { mergePages: false })
  const pages = Array.isArray(extracted.text) ? extracted.text : [extracted.text]
  if (pages.length === 0 || pages.some((page) => page.trim().length === 0)) {
    return { failed: "no_text_layer", bytes: "" }
  }
  return { text: pages.join("\n") }
}

export async function admit(input: {
  locationId: string
  path: string
}): Promise<Source | { refused: "atlas" | "permission" }> {
  const store = requireStore()
  const placed = gate(store, input.locationId, input.path)
  if ("refused" in placed) return placed
  const read = await store.readBytes(placed.path)
  if ("failed" in read && read.failed === "missing") throw new Error("rejected")
  const bytes = "text" in read ? read.text : read.bytes
  const hash = digest(bytes)
  const text = "text" in read ? read.text : ""
  const existing = store.sources().find((row) => row.locationId === input.locationId && row.path === placed.path)
  // Same bytes keep the stored status, including read. Do not insert a second row.
  if (existing && existing.contentHash === hash) return publicSource(existing)
  const row = {
    id: existing ? existing.id : `src_${digest(input.locationId + "\0" + placed.path).slice(0, 12)}`,
    locationId: input.locationId,
    path: placed.path,
    contentHash: hash,
    status: "not_yet_read" as const,
    text,
  }
  store.putSource(row)
  return publicSource(row)
