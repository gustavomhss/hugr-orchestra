export function nativeRecord(value: unknown, label: string): Record<string, unknown> {
  if (typeof value !== "object" || value === null || Array.isArray(value)) throw new Error(`native-object: ${label}`)
  return value as Record<string, unknown>
}

export function nativeKeys(value: Record<string, unknown>, expected: readonly string[], label: string) {
  if (Object.keys(value).length !== expected.length || !expected.every((key) => Object.hasOwn(value, key)))
    throw new Error(`native-fields: ${label}`)
}

export function nativeText(value: unknown, label: string): string {
  if (typeof value !== "string" || !value.trim() || value.normalize("NFC") !== value)
    throw new Error(`native-text: ${label}`)
  return value
}

export function nativeStrings(value: unknown, label: string, unique = true): string[] {
  if (!Array.isArray(value)) throw new Error(`native-array: ${label}`)
  const result = value.map((item) => nativeText(item, label))
  if (unique && new Set(result).size !== result.length) throw new Error(`native-duplicate: ${label}`)
  return result
}

export function nativeDigest(value: unknown, length: 40 | 64, label: string): string {
  if (typeof value !== "string" || !new RegExp(`^[0-9a-f]{${length}}$`).test(value)) throw new Error(`native-digest: ${label}`)
  return value
}

export function nativePath(value: unknown): string {
  const path = nativeText(value, "path")
  if (/[\\:*?"<>|\u0000-\u001f\u007f]/.test(path) || path.split("/").some((part) => !part || part === "." || part === ".." || [".git", ".atlas"].includes(part.toLowerCase()) || /[. ]$/.test(part)))
    throw new Error(`native-path: ${path}`)
  return path
}

export function nativeRoot(path: string): string {
  const root = resolve(path)
  const stat = lstatSync(root)
  if (!stat.isDirectory() || stat.isSymbolicLink() || realpathSync(root) !== root) throw new Error(`native-root-path: ${path}`)
  return root
}

export function readNativeFile(root: string, relative: string, maximum = nativeMaxBytes): Uint8Array {
  nativePath(relative)
  const parts = relative.split("/")
  const observed = parts.map((_, index) => {
    const path = resolve(root, ...parts.slice(0, index + 1))
    const stat = lstatSync(path)
    if (stat.isSymbolicLink() || (index < parts.length - 1 ? !stat.isDirectory() : !stat.isFile())) throw new Error(`native-file-kind: ${relative}`)
    return { path, stat }
  })
  const last = observed.at(-1)!
  if (last.stat.size > maximum) throw new Error(`native-file-limit: ${relative}`)
  const fd = openSync(last.path, constants.O_RDONLY | (constants.O_NOFOLLOW ?? 0) | (constants.O_NONBLOCK ?? 0))
  try {
    const before = fstatSync(fd)
    if (!before.isFile() || before.ino !== last.stat.ino || before.dev !== last.stat.dev || before.size > maximum) throw new Error(`native-file-changed: ${relative}`)
    const chunks: Buffer[] = []
    const buffer = Buffer.alloc(8192)
    let length = 0
    for (;;) {
      const count = readSync(fd, buffer, 0, buffer.length, null)
      if (!count) break
      length += count
      if (length > maximum) throw new Error(`native-file-limit: ${relative}`)
      chunks.push(Buffer.from(buffer.subarray(0, count)))
    }
    const after = fstatSync(fd)
    if (after.size !== before.size || after.mtimeMs !== before.mtimeMs || after.ctimeMs !== before.ctimeMs || length !== after.size)
      throw new Error(`native-file-changed: ${relative}`)
    for (const entry of observed) {
      const stat = lstatSync(entry.path)
      if (stat.isSymbolicLink() || stat.ino !== entry.stat.ino || stat.dev !== entry.stat.dev || stat.isDirectory() !== entry.stat.isDirectory())
        throw new Error(`native-file-changed: ${relative}`)
    }
    return Buffer.concat(chunks, length)
  } finally { closeSync(fd) }
}
