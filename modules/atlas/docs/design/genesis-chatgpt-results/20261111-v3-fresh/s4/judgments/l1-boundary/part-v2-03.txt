    const dependencies = { npm: yield* Npm.Service, processes: yield* AppProcess.Service }
    const client = yield* LLMClient.Service
    const unavailable = (error: Atlas.Unavailable) =>
      new ServiceUnavailableError({ service: "atlas", message: error.message })
    const at = <A, R>(call: (atlas: Atlas.Interface) => Effect.Effect<A, Atlas.Unavailable, R>) =>
      response(
        Effect.gen(function* () {
          const location = yield* Location.Service
          return yield* call(
            Atlas.open(
              dependencies,
              Atlas.storeRoot(location.project.directory, location.directory),
              location.project.id,
            ),
          )
        }).pipe(Effect.mapError(unavailable)),
      )
    // One plain call to the chosen model; `system` keeps the model off its general coding prompt. The Location's catalog
    // and integrations are read per request, so the returned Effect needs nothing more.
    const model = Effect.fn(function* (providerID: string, modelID: string, system: string) {
      const catalog = yield* Catalog.Service
      const integrations = yield* Integration.Service
      return Effect.gen(function* () {
        const selected = (yield* catalog.model.available()).find(
          (item) => item.providerID === providerID && item.id === modelID,
        )
        if (!selected) return yield* Effect.fail(new Error(`Model unavailable: ${providerID}/${modelID}`))
        const resolved = yield* SessionRunnerModel.fromCatalogModel(
          selected,
          yield* SessionRunnerModel.credential(catalog, integrations, selected),
        )
        return (prompt: string) => {
          // Gateways such as OpenCode Go refuse a request without a session to route it by; each call is its own.
          const sessionID = SessionID.descending()
          return client
            .stream(
              LLM.request({
                model: resolved,
                http: { headers: SessionRunnerModel.sessionHeaders(resolved, sessionID) },
                system: [SystemPart.make(system)],
                prompt,
              }),
            )
            .pipe(
              Stream.filter(LLMEvent.is.textDelta),
              Stream.map((event) => event.text),
              Stream.mkString,
            )
        }

export const delegate = Effect.fn("AtlasDelegate.delegate")(function* (
  services: DelegateServices,
  input: DelegateInput,
) {
  const review = yield* services.atlas.review()
  if (review.state === "empty") return EMPTY as DelegateResult
  const drafted = yield* Effect.forEach(input.questionIds, (questionId) => {
    const question = review.interview.batch.find((item) => item.id === questionId)
    if (!question) return Effect.succeed(refused(questionId, "not an open question in the current interview batch"))
    return draftOne(services, input, question).pipe(
      Effect.catch((error) => Effect.succeed(refused(questionId, firstLine(error)))),
    )
  })
  return { state: "ok", drafted } as DelegateResult
})

const draftOne = Effect.fn("AtlasDelegate.draftOne")(function* (
  services: DelegateServices,
  input: DelegateInput,
  question: AtlasInterviewQuestion,
) {
  const reply = yield* services.draft(delegatePrompt(question))
  if (!reply || !reply.answer.trim())
    return refused(question.id, "the model did not reply with a non-empty answer and why")
  const written = yield* services.atlas.answer({
    questionId: question.id,
    answer: reply.answer.trim(),
    drafted: { model: `${input.providerID}/${input.modelID}`, why: reply.why.trim() },
  })
  if (written.state === "empty") return refused(question.id, "the Atlas store is gone")
  if (!written.ok) return refused(question.id, written.rejected ?? "the surface refused the draft")
  return { questionId: question.id, ok: true, rejected: null }
