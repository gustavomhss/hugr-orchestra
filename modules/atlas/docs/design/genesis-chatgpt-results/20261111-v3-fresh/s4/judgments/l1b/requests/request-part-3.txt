## Continuity: recoverable request reduction around the Session core

**Trigger, fallback, and lifetime.** Continuity defaults on, with a background trigger of 0.4 of the provider context window; only finite configured triggers strictly between 0 and 0.7 are accepted. The hard input threshold is the smaller of 70% of context and context minus reserved output, floored and bounded below by zero; an unknown/nonpositive context gives Infinity. export async function retrievalProof(questionId: string, revisionId: string): Promise<boolean> {
  const store = requireStore()
  const question = store.questions().find((row) => row.id === questionId)
  if (!question) return false
  if (question.revisionId !== revisionId) return false
  const chunk = store
    .chunks()
    .find((item) => item.sourceId === question.span.sourceId && item.start === question.span.start && item.end === question.span.end)
  if (!chunk) return false
  const trace = store.trace(questionId)
  if (!trace) return false
  const revision = store.revision(revisionId)
  if (trace.k !== undefined && revision?.k !== undefined && trace.k !== revision.k) throw new Error("rejected")
  return promptChunkIds(trace).includes(chunk.id)
}

export function fenced(text: string, language = "text") {
  const fence = "`".repeat(Array.from(text.matchAll(/`+/g)).reduce((size, match) => Math.max(size, match[0].length + 1), 3))
  // The extra LF is framing, not part of the captured payload (including CRLF).
  return `${fence}${language}\n${text}\n${fence}`
}

/** Splits text into UTF-8 payloads of at most PAYLOAD_LIMIT bytes without cutting a code point. */
export function split(text: string) {
  const bytes = Buffer.from(text.toWellFormed(), "utf8")
  const result: string[] = []
  for (let start = 0; start < bytes.length; ) {
    let end = Math.min(start + PAYLOAD_LIMIT, bytes.length)
    while (end < bytes.length && (bytes[end] & 0xc0) === 0x80) end--
    result.push(bytes.subarray(start, end).toString("utf8"))
    start = end
  }
  return result
}
