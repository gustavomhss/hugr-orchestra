import type { Pipeline, PipelineField, Revision } from "./types"
import { digest, requireStore } from "./context"
import type { Fill, StoredRevision } from "./context"

export function readPipeline(locationId: string): Promise<Pipeline> {
  return Promise.resolve(requireStore().pipeline(locationId))
}

export function setPipeline(locationId: string, field: PipelineField, value: string | number): Promise<Pipeline> {
  const store = requireStore()
  const current = store.pipeline(locationId)
  const next = {
    ...current,
    [field]: field === "k" ? Number(value) : String(value),
  } as Pipeline
  store.setPipeline(locationId, next)
  return Promise.resolve(next)
}

export async function readRevision(revisionId: string): Promise<Revision> {
  const found = requireStore().revision(revisionId)
  if (!found) throw new Error("rejected")
  return {
    id: found.id,
    snapshot: found.snapshot,
    pipelineHash: found.pipelineHash,
    fill: found.fill,
  }
}

export function pipelineHash(pipeline: Pipeline) {
  return digest(
    JSON.stringify({
      boundary: pipeline.boundary,
      k: pipeline.k,
      parser: pipeline.parser,
      tokenizer: pipeline.tokenizer,
    }),
  )
}

export function openRevision(locationId: string): Promise<StoredRevision> {
  const store = requireStore()
  const pipeline = store.pipeline(locationId)
  const members = store
    .sources()
    .filter((row) => row.locationId === locationId)
    .map((row) => ({ sourceId: row.id, contentHash: row.contentHash }))
    .sort((left, right) => (left.sourceId < right.sourceId ? -1 : left.sourceId > right.sourceId ? 1 : 0))
  const revision: StoredRevision = {
    id: store.allocate("rev"),
    snapshot: digest(members.map((row) => `${row.sourceId}\t${row.contentHash}`).join("\n")),
    pipelineHash: pipelineHash(pipeline),
    fill: fillFor(locationId, members),
    locationId,
    members,
    parser: pipeline.parser,
    boundary: pipeline.boundary,
    tokenizer: pipeline.tokenizer,
    k: pipeline.k,
  }
  store.putRevision(revision)
  return Promise.resolve(revision)
}

export function fillFor(locationId: string, members: Array<{ sourceId: string; contentHash: string }>): Fill {
  const store = requireStore()
  if (store.failed(locationId)) return "failed"
  const memberIds = new Set(members.map((row) => row.sourceId))
  const stale = store.chunks().some((chunk) => {
    const source = store.sources().find((row) => row.id === chunk.sourceId)
    if (!source) return true
    if (source.locationId !== locationId) return false
    if (!memberIds.has(source.id)) return true
    return source.contentHash !== chunk.sourceHash
  })
  if (stale) return "failed"
  return "complete"
}

import type { CompareRow } from "./types"
import { requireStore } from "./context"
import type { SpanRef, StoredRevision } from "./context"

export function compareRevisions(
  leftId: string,
  rightId: string,
  proofSetId: string,
): Promise<{ rejected: true } | { questions: CompareRow[] }> {
  const store = requireStore()
  const left = store.revision(leftId)
  const right = store.revision(rightId)
  if (!left || !right) return Promise.resolve({ rejected: true })
  if (proofSetId !== left.snapshot || left.snapshot !== right.snapshot) return Promise.resolve({ rejected: true })
  const stages = ["parser", "boundary", "tokenizer", "k"] as const
  const changed = stages.filter((stage) => left[stage] !== right[stage])
  if (changed.length !== 1) return Promise.resolve({ rejected: true })
  const recut = store.chunks().some((chunk) => store.chunkCut(chunk.id) === right.boundary)
  const questions = store.questions().map((question) => {
    if (!recut || right.boundary === left.boundary) {
      return { id: question.id, startChanged: false, endChanged: false, textChanged: false }
    }
    const after = resolve(right, question.text)
    return {
      id: question.id,
      startChanged: question.span.start !== after.start,
      endChanged: question.span.end !== after.end,
      textChanged: textOf(question.span) !== textOf(after),
    }
  })
  return Promise.resolve({ questions })
}

function resolve(revision: StoredRevision, sentence: string): SpanRef {
  const store = requireStore()
  const hit = store.chunks().find((chunk) => {
    const source = store.sources().find((row) => row.id === chunk.sourceId)
    return source?.locationId === revision.locationId && store.chunkCut(chunk.id) === revision.boundary && chunk.text.includes(sentence)
  })
  if (hit) return { sourceId: hit.sourceId, start: hit.start, end: hit.end }
  const source = store.sources().find((row) => row.locationId === revision.locationId && row.text.includes(sentence))
  if (!source) return { sourceId: "", start: 0, end: 0 }
  const start = source.text.indexOf(sentence)
  return { sourceId: source.id, start, end: start + sentence.length }
}

function textOf(span: SpanRef) {
  const source = requireStore().sources().find((row) => row.id === span.sourceId)
  if (!source) return ""
  return source.text.slice(span.start, span.end)
}
