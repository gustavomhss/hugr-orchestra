```ts
// Pure formatting helpers from specs/atlas-ui/mock/ui.js. They return text (or, for `highlight`, escaped HTML).

export const num = (n: number) => n.toLocaleString("en-US")
export const short = (hex: string | null | undefined, n = 8) => (hex ? hex.slice(0, n) : "")
// A stable 16-hex-digit FNV-1a fingerprint of a seed (A.hash).
export function hash(seed: string) {
  const state = { h: 2166136261 }
  return Array.from({ length: 2 }, (_, i) => {
    for (const c of `${seed}:${i}`) state.h = Math.imul(state.h ^ c.charCodeAt(0), 16777619) >>> 0
    return state.h.toString(16).padStart(8, "0")
  }).join("")
}
// The display name of a seat: its id with a capital first letter; actor ids with a colon stay as they are.
export const seatName = (id: string) => (id.includes(":") ? id : id.charAt(0).toUpperCase() + id.slice(1))

const ESCAPES: Record<string, string> = { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }
export const esc = (value: string) => value.replace(/[&<>"']/g, (c) => ESCAPES[c])
// Tiny TypeScript-ish highlighter for code previews (keywords, strings, comments, types). It escapes first, so its
// output is safe for innerHTML.
export const highlight = (line: string) =>
  esc(line)
    .replace(/(\/\/.*)$/, '<span class="tk-c">$1</span>')
    .replace(/(&quot;.*?&quot;|`.*?`)/g, '<span class="tk-s">$1</span>')
    .replace(/\b(export|const|function|return|if|yield|import|from|type|await|async|new|throw|readonly)\b/g, '<span class="tk-k">$1</span>')
    .replace(/\b([A-Z][A-Za-z0-9]+)\b/g, '<span class="tk-t">$1</span>')
```

```ts
import { describe, expect, test } from "bun:test"
import { esc, hash, highlight, num, seatName, short } from "./format"

describe("atlas kit format", () => {
  test("formats numbers and short hashes", () => {
    expect(num(1234567)).toBe("1,234,567")
    expect(short("abcdef0123", 4)).toBe("abcd")
    expect(short("abcdef0123")).toBe("abcdef01")
    expect(short(null)).toBe("")
  })

  test("fingerprints a seed the same way as the mock", () => {
    expect(hash("x")).toBe("88017b157ee0fc92")
    expect(hash("packages/a")).toBe("37ededc6267c5ece")
  })

  test("names seats from their ids", () => {
    expect(["backend", "jimmy", "maestro", "you", "seat:orchestrator"].map(seatName)).toEqual(["Backend", "Jimmy", "Maestro", "You", "seat:orchestrator"])
  })

  test("escapes before highlighting", () => {
    expect(esc(`<a href="x">'&'</a>`)).toBe("&lt;a href=&quot;x&quot;&gt;&#39;&amp;&#39;&lt;/a&gt;")
    expect(highlight('export const A = "x" // <b>')).toBe(

    )
    expect(highlight("type Store = Map")).toBe('<span class="tk-k">type</span> <span class="tk-t">Store</span> = <span class="tk-t">Map</span>')
  })
})
```

```tsx
  return {
    familyWord: (family: string) => {
      const copy = FAMILIES[family]
      return copy ? language.t(copy[0]) : family
    },
    origin: (fact: { scope: string }) =>
      language.t(fact.scope === "atlas:mined" ? "orchestra.atlas.kit.origin.mined" : "orchestra.atlas.kit.origin.written"),
    refusal: (code: string) => {
      const key = REFUSALS[code]
      return key ? language.t(key) : code
    },
    requirement: (id: AtlasReq) => language.t(REQS[id]),
    remedy: (gate: AtlasGate) => (gate.remedy ? language.t(REMEDIES[gate.remedy], { scope: gate.scope ?? "" }) : ""),
  }
}
```
