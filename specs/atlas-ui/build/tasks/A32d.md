2. Create `packages/app-shell/src/orchestra/chapters/atlas/sheets/promote-data.ts` with exactly:
```ts
import type { AtlasPromoteRecord } from "./state"

// The promote result sheet's counts (mock app.js:202-205): new facts (accepted rows that were staged), facts
// re-checked (accepted rows that were already facts), refused suggestions, and every refused row grouped by its
// refusal code, in first-seen order.
export function promoteSummary(record: AtlasPromoteRecord) {
  const rows = record.run.rows
  const staged = (key: string) => record.staged.includes(key)
  const refused = rows.filter((row) => !row.settled)
  const code = (row: { rejected?: string }) => row.rejected ?? "unknown"
  return {
    fresh: rows.filter((row) => row.settled && staged(row.nodeKey)).length,
    again: rows.filter((row) => row.settled && !staged(row.nodeKey)).length,
    refusedNew: refused.filter((row) => staged(row.nodeKey)).length,
    groups: [...new Set(refused.map(code))].map((reason) => ({
      code: reason,
      keys: refused.filter((row) => code(row) === reason).map((row) => row.nodeKey),
    })),
  }
}
```

```ts
import { describe, expect, test } from "bun:test"
import { promoteSummary } from "./promote-data"
import type { AtlasPromoteRecord } from "./state"

const run = (rows: AtlasPromoteRecord["run"]["rows"]): AtlasPromoteRecord["run"] => ({
  state: "ok",
  read: true,
  candidates: rows.length,
  promoted: rows.filter((row) => row.settled).length,
  refused: rows.filter((row) => !row.settled).length,
  rows,
})

describe("atlas promote summary", () => {
  test("splits new facts from re-checked ones and groups refusals by reason", () => {
    const record: AtlasPromoteRecord = {
      run: run([
        { nodeKey: "k5", settled: true },
        { nodeKey: "k1", settled: true },
        { nodeKey: "k6", settled: false, rejected: "ungrounded" },
        { nodeKey: "k2", settled: false, rejected: "ungrounded" },
        { nodeKey: "k3", settled: false, rejected: "governance-downgrade" },
      ]),
      staged: ["k5", "k6"],
    }
    expect(promoteSummary(record)).toEqual({
      fresh: 1,
      again: 1,
      refusedNew: 1,
      groups: [
        { code: "ungrounded", keys: ["k6", "k2"] },
        { code: "governance-downgrade", keys: ["k3"] },
      ],
    })
  })

    expect(promoteSummary({ run: run([{ nodeKey: "a", settled: false }]), staged: [] })).toEqual({

      again: 0,
      refusedNew: 0,
      groups: [{ code: "unknown", keys: ["a"] }],

  })
```

```tsx
export function PromoteAllSheet(props: AtlasSheetProps<"promote-all">) {
  const atlas = useAtlas()
  const language = useLanguage()
  const [form, setForm] = createStore({ token: "" })
  const waiting = () => atlas.model()?.candidates.filter((x) => !x.promotedNow).length ?? 0
  const run = async () => {
    // The keys staged before the run tell the result sheet which accepted rows are new facts.
    const staged = atlas.model()?.candidates.map((x) => x.key) ?? []
    const result = await atlas.controller.promote({ token: form.token })
    if (!result || isAtlasEmpty(result)) return
    keepPromoteRun({ run: result, staged })
    openAtlasSheet({ kind: "promote-result" })
  }
  return (
    <AtDialog
      title={language.t("orchestra.atlas.sheets.promoteAll.title")}
      text={language.plural("orchestra.atlas.sheets.promoteAll.text", waiting(), { n: num(waiting()) })}
      onClose={props.onClose}
      footer={
        <AtDialogFoot
          label={language.t("orchestra.atlas.sheets.promoteAll.confirm")}
          disabled={atlas.controller.state.pending.includes("promote")}
          onCancel={props.onClose}
          onConfirm={() => void run()}
        />
      }
    >
      <WritingAs token={form.token} onToken={(token) => setForm("token", token)} />
      <LockedNote />
      <p class="mx-note">{language.t("orchestra.atlas.sheets.promoteAll.note")}</p>
    </AtDialog>
  )
}
```
