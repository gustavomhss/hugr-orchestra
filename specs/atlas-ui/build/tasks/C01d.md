   ```ts
   // @atlas/adapter-io — src/surface/types-review.ts  (the Atlas UI surface: the review queue — D31, D35)
   //
   // The shapes `openSurface().review()` returns and the review writes accept: held candidates (R2), open ratification
   // proposals (R3), open memory proposals (R1), the interview batch with its model drafts (R11), a territory
   // declaration (R4) and the host-only interview delegation (D33). Plain JSON; an absent value is `null`, never an
   // omitted key (lead decision D35), except optional INPUT fields. Every engine shape here is MIRRORED field for
   // field, not imported, and this file imports nothing, for the reason ./types.ts gives. The result unions that need
   // `AtlasEmpty`/`AtlasVerdict` live in ./types.ts, which imports this file.

   /** One open ratification proposal (`ReviewOf<"proposal">`). */
   export interface AtlasReviewProposal {
     readonly id: string
     readonly nodeKey: string
     readonly toTier: "T1" | "T0"
     readonly reason: string
     readonly by: string
     readonly at: string
   }

   /** One open memory proposal (`ReviewOf<"memory-proposal">`): a seat's project rule waiting for a person. */
   export interface AtlasMemoryProposal {
     readonly id: string
     readonly member: string
     readonly rule: string
     readonly scope: string
     readonly by: string
     readonly at: string
   }

   /** A model's drafted answer (`InterviewQuestion.draft`). The question stays open until a person answers it. */
   export interface AtlasInterviewDraft {
     readonly answer: string
     readonly model: string
     readonly why: string
     readonly by: string
     readonly at: string
   }

   /** One interview question (`InterviewQuestion`). `site` is the anchored path (`OpenQ.site.qualifiedPath`). */
   export interface AtlasInterviewQuestion {
     readonly id: string
     readonly kind: "owner" | "tier" | "contested" | "intent"
     readonly subject: string
     readonly ask: string
     readonly rank: number
     readonly site: string
     readonly options: readonly string[] | null
     readonly nodeKey: string | null
     readonly draft: AtlasInterviewDraft | null
   }

   /** `review()`. `unreadable`: the review log has lines that do not verify, or cannot be read at all — what did verify
    *  is still served, and every review write refuses until the log is repaired. */
   export interface AtlasReview {
     readonly state: "ok"
     readonly unreadable: boolean
     /** The staged candidates' nodeKeys every `promote` skips, sorted. */
     readonly held: readonly string[]
     readonly proposals: readonly AtlasReviewProposal[]
     readonly memoryProposals: readonly AtlasMemoryProposal[]
     readonly interview: { readonly batch: readonly AtlasInterviewQuestion[]; readonly deferred: number }
   }

   /** What a review write returns as `data`: the appended record's id (the catalog path for `territory`). */
   export interface AtlasReviewWriteOut {
     readonly id: string
   }

   /** `atlas hold <nodeKey>`. */
   export interface AtlasHoldInput {
     readonly nodeKey: string
   }

   /** `atlas release <nodeKey>`. */
   export interface AtlasReleaseInput {
     readonly nodeKey: string
   }

   /** `atlas propose <nodeKey> <T1|T0> <reason...>`. */
   export interface AtlasProposeInput {
     readonly nodeKey: string
     readonly toTier: "T1" | "T0"
     readonly reason: string
   }

   /** `atlas decide <proposalId> ratify|reject <reason...>`. `token` is the ratify token for a ratification. */
   export interface AtlasDecideInput {
     readonly proposalId: string
     readonly verdict: "ratified" | "rejected"
     readonly reason: string
     readonly token?: string
   }

   /** `atlas answer <questionId> <answer...>`. `drafted` marks a model's draft, which never proposes. */
   export interface AtlasAnswerInput {
     readonly questionId: string
     readonly answer: string
     readonly drafted?: { readonly model: string; readonly why: string }
   }

   /** A person's decision on a memory proposal. A non-empty `rule`/`scope` replaces the proposal's text. */
   export interface AtlasMemoryDecideInput {
     readonly proposalId: string
     readonly verdict: "indexed" | "discarded"
     readonly rule?: string
     readonly scope?: string
   }

   /** `atlas territory <name> <glob...> --owner --tier [--project]`. `projectId` is required only for the first catalog
    *  (the Orchestra project id; issues/c-engine.md item 2). `token` is the ratify token, else the process's own. */
   export interface AtlasTerritoryInput {
     readonly name: string
     readonly owner: string
     readonly tier: "T0" | "T1" | "T2"
     readonly globs: readonly string[]
     readonly token?: string
     readonly projectId?: string
   }

   /** The host-only interview delegation (D33): one model drafts an answer per question id. */
   export interface AtlasDelegateInput {
     readonly questionIds: readonly string[]
     readonly providerID: string
     readonly modelID: string
   }

   /** What the delegation drafted: one row per question id, `rejected` naming why a draft was not written. */
   export interface AtlasDelegateOut {
     readonly state: "ok"
     readonly drafted: readonly { readonly questionId: string; readonly ok: boolean; readonly rejected: string | null }[]
   }
   ```
2. Create `packages/adapter-io/src/surface/review.ts`:
   ```ts
   // @atlas/adapter-io — src/surface/review.ts  (the Atlas UI surface: the review queue — D31)
   //
   // `review()` reads the shared review log (`../review-log.ts`, D14) and the interview (`interviewOf`, D29): what is
   // held (R2), the open ratification proposals (R3) and memory proposals (R1), and the interview batch with a model's
   // draft riding along (R11). The review writes act through the same adapter-io helpers the CLI's review commands
   // call, so the screen and the CLI cannot diverge, and they refuse with the CLI's own words.
   //
   // COST, STATED: `review()` builds the code index once (`openQuestionsOf`), like `atlas interview`.

   import { interviewOf } from "../open-questions-source.js"
   import type { InterviewQuestion } from "../open-questions-source.js"
   import { foldReview, readReview } from "../review-log.js"
   import type { AtlasInterviewQuestion, AtlasReview } from "./types-review.js"

   /** The review queue at `repoPath`. A log that cannot be fully read serves what verified and says so. */
   export function review(repoPath: string): AtlasReview {
     const read = readReview(repoPath)
     const fold = foldReview(read.records)
     const interview = interviewOf(repoPath)
     return {
       state: "ok",
       unreadable: read.unreadable === true || read.rejected > 0,
       held: [...fold.held].sort(),
       proposals: fold.openProposals.map((p) => ({
         id: p.id,
         nodeKey: p.nodeKey,
         toTier: p.toTier,
         reason: p.reason,
         by: p.by,
         at: p.at,
       })),
       memoryProposals: fold.openMemoryProposals.map((p) => ({
         id: p.id,
         member: p.member,
         rule: p.rule,
         scope: p.scope,
         by: p.by,
         at: p.at,
       })),
       interview: { batch: interview.batch.map(question), deferred: interview.deferred },
     }
   }

   function question(q: InterviewQuestion): AtlasInterviewQuestion {
     return {
       id: q.id,
       kind: q.kind,
       subject: q.subject,
       ask: q.ask,
       rank: q.rank,
       site: q.site.qualifiedPath,
       options: q.options === undefined ? null : [...q.options],
       nodeKey: q.nodeKey ?? null,
       draft:
         q.draft === undefined
           ? null
           : { answer: q.draft.answer, model: q.draft.model, why: q.draft.why, by: q.draft.by, at: q.draft.at },
     }
   }
   ```

   ```ts
   // @atlas/adapter-io — test/surface-review.test.ts  (D31: the Atlas UI surface's review queue, `review()`)
   //
   // The review log folded for the screen (held keys, open proposals, open memory proposals), a log with a line that does
   // not verify, and the interview batch over a real store (open-questions-source.test.ts's fixture) with a model's
   // draft riding along until a person answers.

   import { appendFileSync, mkdirSync, mkdtempSync, rmSync, writeFileSync } from "node:fs"
   import { tmpdir } from "node:os"
   import { join } from "node:path"
   import { afterEach, describe, expect, it } from "vitest"
   import { emptyStore, upsert } from "@atlas/knowledge"
   import type { PredicateSlot } from "@atlas/knowledge"
   import { appendReview, reviewLogPath } from "../src/review-log.js"
   import { createDiskStore } from "../src/store.js"
   import { openSurface } from "../src/surface/index.js"
   import { makeFixRepo } from "./harness/fix-repo.js"

   const cleanups: (() => void)[] = []
   afterEach(() => {
     for (const cleanup of cleanups.splice(0)) cleanup()
   })

   const NOW = "2026-10-18T10:00:00.000Z"
   const LATER = "2026-10-18T11:00:00.000Z"

   /** A temporary directory with an empty `.atlas/`, so the surface answers `ok`. */
   function atlasDir(): string {
     const dir = mkdtempSync(join(tmpdir(), "atlas-surface-review-"))
     cleanups.push(() => rmSync(dir, { recursive: true, force: true }))
     mkdirSync(join(dir, ".atlas"), { recursive: true })
     return dir
   }

   /** open-questions-source.test.ts's store: a T2 invariant in packages/a (a tier question, plus the unit's owner and
    *  intent questions) and a rationale in packages/b, whose policy owner is alice. */
   function questionRepo(): string {
     const repo = makeFixRepo()
     cleanups.push(repo.cleanup)
     mkdirSync(join(repo.repoPath, ".atlas"), { recursive: true })
     writeFileSync(
       join(repo.repoPath, ".atlas", "policy.json"),
       '{"t0Heuristic":{"keywords":[]},"authz":{"scopes":{"packages/b":["alice"]}}}',
     )
     const store = createDiskStore(join(repo.repoPath, ".atlas", "cas"))
     const rows: readonly { nodeKey: string; claimNorm: string; anchor: string; slot: PredicateSlot }[] = [
       { nodeKey: "k1", claimNorm: "x holds", anchor: "packages/a/src/x.ts::f", slot: "invariant" },
       { nodeKey: "k3", claimNorm: "b is for z", anchor: "packages/b/src/z.ts", slot: "rationale" },
     ]
     let projection = emptyStore()
     for (const row of rows) {
       const contentHash = store.put({ kind: "advisory", tier: "T2", claimNorm: row.claimNorm, authoring: "ADVISORY" } as never) as string
       projection = upsert(projection, {
         nodeKey: row.nodeKey,
         contentHash,
         family: "advisory",
         claimNorm: row.claimNorm,
         primaryAnchor: row.anchor,
         slot: row.slot,
         tier: "T2",
       }).store
     }
     store.persistProjection(projection)
     return repo.repoPath
   }

   /** Narrow a surface result to its `state: "ok"` member, failing the test on any other state. */
   function ok<T extends { readonly state: string }>(value: T): Extract<T, { readonly state: "ok" }> {
     if (value.state !== "ok") throw new Error(`expected state ok, got ${value.state}`)
     return value as Extract<T, { readonly state: "ok" }>
   }

   describe("openSurface().review — the review queue the screen works through", () => {
     it("answers empty when the repository has no .atlas directory", () => {
       const dir = mkdtempSync(join(tmpdir(), "atlas-surface-review-empty-"))
       cleanups.push(() => rmSync(dir, { recursive: true, force: true }))
       expect(openSurface(dir).review()).toEqual({ state: "empty" })
     })

     it("folds the log: held keys sorted, open proposals and open memory proposals only", () => {
       const dir = atlasDir()
       appendReview(dir, { kind: "hold", nodeKey: "nk-b", at: NOW, by: "alice" })
       appendReview(dir, { kind: "hold", nodeKey: "nk-a", at: NOW, by: "alice" })
       appendReview(dir, { kind: "hold", nodeKey: "nk-z", at: NOW, by: "alice" })
       appendReview(dir, { kind: "release", nodeKey: "nk-z", at: LATER, by: "alice" })
       const open = appendReview(dir, { kind: "proposal", nodeKey: "nk-c", toTier: "T1", reason: "it holds", at: NOW, by: "lucy" })
       const closed = appendReview(dir, { kind: "proposal", nodeKey: "nk-d", toTier: "T0", reason: "core", at: NOW, by: "lucy" })
       appendReview(dir, { kind: "decision", proposalId: closed.id, verdict: "rejected", reason: "no", at: LATER, by: "gus" })
       const rule = appendReview(dir, {
         kind: "memory-proposal",
         member: "backend",
         rule: "run go vet first",
         scope: "packages/api",
         at: NOW,
         by: "backend",
       })
       expect(openSurface(dir).review()).toEqual({
         state: "ok",
         unreadable: false,
         held: ["nk-a", "nk-b"],
         proposals: [{ id: open.id, nodeKey: "nk-c", toTier: "T1", reason: "it holds", by: "lucy", at: NOW }],
         memoryProposals: [{ id: rule.id, member: "backend", rule: "run go vet first", scope: "packages/api", by: "backend", at: NOW }],
         interview: { batch: [], deferred: 0 },
       })
     })

     it("says the log is unreadable when a line does not verify, and still serves what did", () => {
       const dir = atlasDir()
       appendReview(dir, { kind: "hold", nodeKey: "nk-a", at: NOW, by: "alice" })
       appendFileSync(reviewLogPath(dir), "{torn\n")
       const queue = ok(openSurface(dir).review())
       expect([queue.unreadable, queue.held]).toEqual([true, ["nk-a"]])
     })

     it("serves the interview batch with null for absent values, and a model's draft until a person answers", () => {
       const repoPath = questionRepo()
       const surface = openSurface(repoPath)
       const first = ok(surface.review()).interview
       expect(first.deferred).toBe(0)
       expect(first.batch.map((q) => `${q.kind} ${q.subject}`).sort()).toEqual(["intent packages/a", "owner packages/a", "tier k1"])
       const tier = first.batch.find((q) => q.kind === "tier")
       if (tier === undefined) throw new Error("expected a tier question")
       expect(tier).toEqual({
         id: tier.id,
         kind: "tier",
         subject: "k1",
         ask: tier.ask,
         rank: 0,
         site: "packages/a/src/x.ts",
         options: ["Rule · T1", "Core · T0", "Keep as a hint"],
         nodeKey: "k1",
         draft: null,
       })
       expect(first.batch.find((q) => q.kind === "intent")?.options).toBeNull()
       expect(first.batch.find((q) => q.kind === "owner")?.nodeKey).toBeNull()
       const drafted = { model: "anthropic/claude", why: "an invariant on a file many depend on" }
       appendReview(repoPath, { kind: "answer", questionId: tier.id, answer: "T1", drafted, at: NOW, by: "alice" })
       const withDraft = ok(surface.review()).interview.batch.find((q) => q.id === tier.id)
       expect(withDraft?.draft).toEqual({ answer: "T1", model: drafted.model, why: drafted.why, by: "alice", at: NOW })
       appendReview(repoPath, { kind: "answer", questionId: tier.id, answer: "T1", at: LATER, by: "alice" })
       expect(ok(surface.review()).interview.batch.some((q) => q.id === tier.id)).toBe(false)
     })
   })
   ```
