   export const REVIEW_INVARIANT =
     "REVIEW-LOG (D14): a review act is one appended, content-keyed record in .atlas/review.jsonl — nothing is edited or removed, and the current state is a fold over the records"

   /**
    * A review act a helper declined. `invalid`: the request itself is wrong (the CLI's exit 1). `governed`: the log or
    * the store declined it (exit 2). `reason` is the exact text the CLI prints and the Atlas screen shows.
    */
   export interface ReviewRefusal {
     readonly ok: false
     readonly refusal: "invalid" | "governed"
     readonly reason: string
   }

   /** The folded review state, or the governed refusal when the log has lines that do not verify: a decision taken over
    *  a log that cannot be fully read could miss the very hold or decision a torn line held. */
   export function foldReviewLog(repoPath: string): { readonly fold: ReviewFold } | ReviewRefusal {
     const read = readReview(repoPath)
     if (read.unreadable !== true && read.rejected === 0) return { fold: foldReview(read.records) }
     const why =
       read.unreadable === true
         ? "could not be read"
         : `has ${read.rejected} line(s) that do not verify (torn, edited, or not a review record)`
     return {
       ok: false,
       refusal: "governed",
       reason: `the review log ${reviewLogPath(repoPath)} ${why}, so a hold or a decision in it could be missed — repair or remove those lines, then re-run`,
     }
   }

   /** `wanted` as an exact id, or as a prefix of at least 8 characters that matches exactly one id; else `undefined`. */
   export function resolveReviewId(ids: readonly string[], wanted: string): string | undefined {
     if (ids.includes(wanted)) return wanted
     const hits = wanted.length >= 8 ? ids.filter((id) => id.startsWith(wanted)) : []
     return hits.length === 1 ? hits[0] : undefined
   }

   // @atlas/adapter-io — src/review-decide.ts  (R3 / D29: decide one open ratification proposal)
   //
   // The decision half of `atlas decide`, moved out of the CLI (lead decision D29) so the CLI runner and the Atlas
   // screen's surface decide through ONE function and cannot diverge. `rejected` only appends the decision, so the
   // rejection stays in the history. `ratified` re-emits the CURRENT fact at the proposed tier through the emit door
   // the CALLER supplies — the composed `atlas-emit` leg, so the truth, authz and KNOW-8 ratify gates every emit faces
   // still apply and the ratify token stays with the composition root — and appends the decision only when the door
   // admitted the re-emit: a refused ratification writes nothing and the proposal stays open.

   import { isRetired } from "@atlas/knowledge"
   import type { GroundedFact } from "@atlas/knowledge"
   import type { Verdict } from "@atlas/tools"
   import { appendReview, foldReviewLog, resolveReviewId } from "./review-log.js"
   import type { ReviewOf, ReviewRefusal } from "./review-log.js"
   import { currentFactOf } from "./review-store.js"
   import type { CurrentFact } from "./review-store.js"

   /** What the emit door answered, reduced to what a decision needs. */
   export interface EmitOutcome {
     readonly emitted: boolean
     readonly id?: string
     readonly rejected?: string
   }

   /** One decision. `proposalId` is an exact id or a unique prefix of at least 8 characters. */
   export interface DecideInput {
     readonly proposalId: string
     readonly verdict: "ratified" | "rejected"
     readonly reason: string
     readonly by: string
     readonly at: string
   }

   export type DecideResult =
     | {
         readonly ok: true
         readonly proposal: ReviewOf<"proposal">
         readonly decision: ReviewOf<"decision">
         /** The re-emitted fact's id, when the door returned one (ratified only). */
         readonly emittedId?: string
       }
     | ReviewRefusal

   /**
    * Decide the open proposal `input.proposalId`. `emit` is the composed `atlas-emit` door, already bound to its ratify
    * token and anchor rev. `factOf` reads the current fact; it defaults to the store at `repoPath`, and exists so a
    * caller's own tests can supply one.
    */
   export function decideProposal(
     repoPath: string,
     input: DecideInput,
     emit: (fact: GroundedFact) => EmitOutcome,
     factOf: (nodeKey: string) => CurrentFact | undefined = (nodeKey) => currentFactOf(repoPath, nodeKey),
   ): DecideResult {
     const reason = input.reason.trim()
     if (reason.length === 0) return { ok: false, refusal: "invalid", reason: "decide requires a reason after the verdict" }
     const read = foldReviewLog(repoPath)
     if ("refusal" in read) return read
     const open = read.fold.openProposals
     const id = resolveReviewId(
       open.map((p) => p.id),
       input.proposalId,
     )
     const proposal = open.find((p) => p.id === id)
     if (proposal === undefined) {
       return {
         ok: false,
         refusal: "invalid",
         reason: `decide: no open proposal matches '${input.proposalId}' — \`atlas proposals\` lists the open ones (a decided proposal is closed)`,
       }
     }
     const decided = (verdict: "ratified" | "rejected") =>
       appendReview(repoPath, {
         kind: "decision",
         proposalId: proposal.id,
         verdict,
         reason,
         at: input.at,
         by: input.by,
       }) as ReviewOf<"decision">
     if (input.verdict === "rejected") return { ok: true, proposal, decision: decided("rejected") }
     const current = factOf(proposal.nodeKey)
     if (current === undefined || isRetired(current.fact)) {
       return {
         ok: false,
         refusal: "governed",
         reason: `ratify: '${proposal.nodeKey}' is no longer a current, live fact, so there is nothing to re-emit; no decision was written — reject the proposal instead`,
       }
     }
     const out = emit({ ...current.fact, tier: proposal.toTier } as GroundedFact)
     if (!out.emitted) {
       return {
         ok: false,
         refusal: "governed",
         reason: `ratify: the emit door refused the ${proposal.toTier} re-emit of '${proposal.nodeKey}' (${out.rejected ?? "no reason given"}); no decision was written and the proposal stays open. A T1 needs ATLAS_RATIFY_TOKEN to name a ratifier; a T0 needs ATLAS_RATIFY_TOKEN=billy`,
       }
     }
     const decision = decided("ratified")
     return out.id === undefined ? { ok: true, proposal, decision } : { ok: true, proposal, decision, emittedId: out.id }
   }

   /** Reduce the composed handler's `atlas-emit` verdict: `data` is the door's `EmitOut` on both the admitted and the
    *  refused path. */
   export function emitOutcomeOf(v: Verdict): EmitOutcome {
     const data = v.data as { readonly emitted?: unknown; readonly id?: unknown; readonly rejected?: unknown } | undefined
     if (data?.emitted === true) return typeof data.id === "string" ? { emitted: true, id: data.id } : { emitted: true }
     const rejected = typeof data?.rejected === "string" ? data.rejected : (v.rejected ?? "the emit door refused without a reason")
     return { emitted: false, rejected }
   }

   // @atlas/cli — src/review-common.ts  (the helpers every review-log command shares)
   //
   // The review log (`@atlas/adapter-io` review-log.ts, D14) is read by `atlas promote` (the held set) and written by
   // the review commands (hold/release, propose/decide, answer). A command that DECIDES from the log must not decide
   // over a log it cannot fully read: a torn or edited line could be the very hold or decision it would miss.
   // Contract: specs/atlas-ui/build/tasks/B-L-API.md §3. The refusal text, the invariant line and the id-prefix rule
   // live in adapter-io (D29), so the Atlas screen shows the same words.

   import { foldReviewLog, REVIEW_INVARIANT } from "@atlas/adapter-io"
   import type { ReviewFold } from "@atlas/adapter-io"
   import { errorVerdict, refusalVerdict } from "./cli-verdict.js"
   import { renderRefusal, renderVerdict } from "./render.js"
   import type { CliVerdict } from "./render.js"

   /** A governed refusal (exit 2): the invocation was well-formed and the review state declined it. */
   export function governedRefusal(message: string): CliVerdict {
     return renderRefusal(refusalVerdict(message))
   }

   /** The folded review state, or the refusal to give when the log has lines that do not verify. */
   export function foldOrRefuse(repoPath: string): { readonly fold: ReviewFold } | { readonly refusal: CliVerdict } {
     const read = foldReviewLog(repoPath)
     return "fold" in read ? read : { refusal: governedRefusal(read.reason) }
   }

   /** A successful review outcome (exit 0) in the CLI's status/next/invariant shape, plus detail lines. */
   export function reviewOutcome(next: string, lines: readonly string[]): CliVerdict {
     return { exitCode: 0, stdout: `${["status: ok", `next: ${next}`, `invariant: ${REVIEW_INVARIANT}`, ...lines].join("\n")}\n` }
   }

   /** A usage error (exit 1): the invocation itself was wrong. */
   export function usageError(message: string): CliVerdict {
     return renderVerdict(errorVerdict(message))
   }

   /** The `--by <actor>` every review write is attributed to, or `undefined` when absent, bare or blank. */
   export function requireBy(flags: Readonly<Record<string, string>>): string | undefined {
     const by = flags["by"]
     return by === undefined || by === "true" || by.trim().length === 0 ? undefined : by.trim()
   }

   // @atlas/cli — src/ratify-cli.ts  (R3: `atlas propose`, `atlas proposals`, `atlas decide`)
   //
   // The ratification queue over the review log (D16). A reviewer PROPOSES a current fact for T1 or T0 with a reason
   // (`--by` names who). A person DECIDES through `decideProposal` (@atlas/adapter-io review-decide.ts, D29), the one
   // function the Atlas screen decides through too: `reject` only appends the decision, so the rejection stays in the
   // history; `ratify` re-emits the fact at the proposed tier THROUGH the composed `atlas-emit` door — the same truth,
   // authz and KNOW-8 ratify gates every emit faces, so `ATLAS_RATIFY_TOKEN` must name a ratifier and a T0 still needs
   // `billy`. A ratification the door refuses writes NO decision: the proposal stays open and the door's own reason is
   // printed.

   import { appendReview, currentFactOf, decideProposal, emitOutcomeOf, headSha } from "@atlas/adapter-io"
   import type { CurrentFact, EmitOutcome } from "@atlas/adapter-io"
   import { isRetired, isWeakerTier } from "@atlas/knowledge"
   import type { GroundedFact } from "@atlas/knowledge"
   import type { CliDeps } from "./cli.js"
   import type { CliVerdict } from "./render.js"
   import { foldOrRefuse, governedRefusal, requireBy, reviewOutcome, usageError } from "./review-common.js"

   /** What propose/proposals/decide read and write through — injected so a test needs no composed runtime. */
   export interface RatifyPorts {
     readonly repoPath: string
     readonly now: () => string
     /** The anchor rev a ratifying re-emit is stamped at (live HEAD). */
     readonly head: () => string
     readonly factOf: (nodeKey: string) => CurrentFact | undefined
     /** The composed `atlas-emit` door; `undefined` when the runtime is not composed. */
     readonly emit: ((node: GroundedFact, at: string) => EmitOutcome) | undefined
   }

   /** The production ports: the repository at cwd, the wall clock, live HEAD, the real store, the composed handler. */
   export function ratifyPorts(deps: CliDeps): RatifyPorts {
     const repoPath = process.cwd()
     const handler = deps.handler
     return {
       repoPath,
       now: () => new Date().toISOString(),
       head: () => headSha(repoPath) ?? "",
       factOf: (nodeKey) => currentFactOf(repoPath, nodeKey),
       emit: handler === undefined ? undefined : (node, at) => emitOutcomeOf(handler.handle("atlas-emit", { node, at })),
     }
   }

   export function runPropose(ports: RatifyPorts, positionals: readonly string[], flags: Readonly<Record<string, string>>): CliVerdict {
     const by = requireBy(flags)
     if (by === undefined) return usageError("propose requires --by <actor>: the reviewer the proposal is attributed to")
     const nodeKey = positionals[0] ?? ""
     const toTier = positionals[1] ?? ""
     const reason = positionals.slice(2).join(" ").trim()
     if (toTier !== "T1" && toTier !== "T0") return usageError(`propose: the target tier must be T1 or T0, got '${toTier}'`)
     if (reason.length === 0) return usageError("propose requires a reason after the tier")
     const current = ports.factOf(nodeKey)
     if (current === undefined) return usageError(`propose: no current fact has nodeKey '${nodeKey}'`)
     if (isRetired(current.fact)) return usageError(`propose: '${nodeKey}' is retired (authoring SUPERSEDED); a retired fact is not proposed`)
     if (!isWeakerTier(current.fact.tier, toTier)) {
       return usageError(`propose: '${nodeKey}' is already ${current.fact.tier}; a proposal must raise the tier`)
     }
     const read = foldOrRefuse(ports.repoPath)
     if ("refusal" in read) return read.refusal
     const open = read.fold.openProposals.find((p) => p.nodeKey === nodeKey && p.toTier === toTier)
     if (open !== undefined) return usageError(`propose: '${nodeKey}' already has an open ${toTier} proposal ${open.id}`)
     const record = appendReview(ports.repoPath, { kind: "proposal", nodeKey, toTier, reason, at: ports.now(), by })
     return reviewOutcome(`proposal ${record.id} waits for a person: \`atlas decide ${record.id} ratify|reject <reason...> --by <you>\``, [
       `proposal: ${record.id} ${nodeKey} ${current.fact.tier} -> ${toTier} by ${by}: ${reason}`,
     ])
   }

   export function runProposals(ports: Pick<RatifyPorts, "repoPath">): CliVerdict {
     const read = foldOrRefuse(ports.repoPath)
     if ("refusal" in read) return read.refusal
     const open = read.fold.openProposals
     return reviewOutcome(
       open.length === 0
         ? "no open proposals — `atlas propose <nodeKey> <T1|T0> <reason...> --by <you>` adds one"
         : `${open.length} open proposal(s) — decide one with \`atlas decide <proposalId> ratify|reject <reason...> --by <you>\``,
       open.map((p) => `  ${p.id} ${p.nodeKey} -> ${p.toTier} proposed by ${p.by} at ${p.at}: ${p.reason}`),
     )
   }

   export function runDecide(ports: RatifyPorts, positionals: readonly string[], flags: Readonly<Record<string, string>>): CliVerdict {
     const by = requireBy(flags)
     if (by === undefined) return usageError("decide requires --by <actor>: the person who decides")
     const wanted = positionals[0] ?? ""
     const verdict = positionals[1] ?? ""
     const reason = positionals.slice(2).join(" ").trim()
     if (verdict !== "ratify" && verdict !== "reject") return usageError(`decide: the verdict must be ratify or reject, got '${verdict}'`)
     if (reason.length === 0) return usageError("decide requires a reason after the verdict")
     const emit = ports.emit
     if (verdict === "ratify" && emit === undefined) {
       return usageError("atlas runtime is not composed yet — the WireConfig seams need the composition-root WP")
     }
     const out = decideProposal(
       ports.repoPath,
       { proposalId: wanted, verdict: verdict === "ratify" ? "ratified" : "rejected", reason, by, at: ports.now() },
       // A rejection never reaches the door, and a ratification without one returned above.
       (fact) => emit?.(fact, ports.head()) ?? { emitted: false, rejected: "atlas runtime is not composed yet" },
       ports.factOf,
     )
     if (!out.ok) return out.refusal === "invalid" ? usageError(out.reason) : governedRefusal(out.reason)
     const p = out.proposal
     const line = `decision: ${out.decision.id} ${out.decision.verdict} ${p.nodeKey} -> ${p.toTier} by ${by}: ${reason}`
     if (out.decision.verdict === "rejected") {
       return reviewOutcome(`proposal ${p.id} rejected; the fact keeps its tier and the rejection stays in the log`, [line])
     }
     return reviewOutcome(`proposal ${p.id} ratified: '${p.nodeKey}' is now ${p.toTier}`, [line, `emitted: ${out.emittedId ?? ""}`])
   }

         reason: `decide: no open proposal matches '${id.slice(0, 4)}' — \`atlas proposals\` lists the open ones (a decided proposal is closed)`,
       })
       expect(decideProposal(tmp, input(id.slice(0, 8), "rejected"), door().emit, () => current("T2")).ok).toBe(true)

       expect(again).toMatchObject({ ok: false, refusal: "invalid" })
     })

     it("refuses a blank reason, and a log with a line that does not verify", () => {
       const id = propose()
       expect(decideProposal(tmp, input(id, "rejected", "  "), door().emit, () => current("T2"))).toEqual({
         ok: false,
         refusal: "invalid",
         reason: "decide requires a reason after the verdict",
       })
       appendFileSync(reviewLogPath(tmp), "{torn\n")
       const torn = decideProposal(tmp, input(id, "rejected"), door().emit, () => current("T2"))

       if (torn.ok) throw new Error("expected a refusal")
       expect(torn.reason).toContain("has 1 line(s) that do not verify")

   })

   describe("emitOutcomeOf", () => {
     const guidance = { next: "n", invariant: "i" }
