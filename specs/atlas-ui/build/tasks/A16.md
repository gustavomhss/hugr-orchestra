import type { DirectorySDK } from "@/context/sdk"
import type { AtlasInput, AtlasOutput } from "./atlas-types"

export class AtlasAPIError extends Error {
  constructor(readonly reason: "unsupported" | "invalid", cause?: unknown) {
    super(reason, { cause })
  }
}

// D5: with no .atlas/ at the instance worktree every read answers { state: "empty" }.
export function isAtlasEmpty(value: unknown): value is { state: "empty" } {
  return typeof value === "object" && value !== null && "state" in value && value.state === "empty"
}

export type AtlasTransport = ReturnType<typeof createAtlasAPI>

// The `atlas` httpapi group (D4) through the generated client, keeping the SDK's directory, auth and error
// interceptors. A server without the group answers 404/405/501, which reads as unsupported.
export function createAtlasAPI(sdk: Pick<DirectorySDK, "client" | "directory" | "protocol">) {
  const response = async <T>(
    run: () => Promise<{ data?: T; error?: unknown; response?: Response }>,
    signal?: AbortSignal,
  ) => {
    const protocol = await sdk.protocol
    signal?.throwIfAborted()
    if (protocol !== "v1") throw new AtlasAPIError("unsupported")
    const result = await run()
    if (result.response && [404, 405, 501].includes(result.response.status)) throw new AtlasAPIError("unsupported")
    if (result.error !== undefined) throw result.error
    if (result.data === undefined) throw new AtlasAPIError("invalid")
    return result.data
  }
  const options = (signal?: AbortSignal) => ({ signal, throwOnError: false as const })
  return {
    snapshot: (signal?: AbortSignal): Promise<AtlasOutput<"snapshot">> =>
      response(() => sdk.client.atlas.snapshot({ directory: sdk.directory }, options(signal)), signal),
    reasoning: (id: string, signal?: AbortSignal): Promise<AtlasOutput<"reasoning">> =>
      response(() => sdk.client.atlas.reasoning({ id, directory: sdk.directory }, options(signal)), signal),
    own: (scope: string, signal?: AbortSignal): Promise<AtlasOutput<"own">> =>
      response(() => sdk.client.atlas.own({ scope, directory: sdk.directory }, options(signal)), signal),
    health: (signal?: AbortSignal): Promise<AtlasOutput<"health">> =>
      response(() => sdk.client.atlas.health({ directory: sdk.directory }, options(signal)), signal),
    promote: (input: AtlasInput<"promote">, signal?: AbortSignal): Promise<AtlasOutput<"promote">> =>
      response(() => sdk.client.atlas.promote({ ...input, directory: sdk.directory }, options(signal)), signal),
    draft: (input: AtlasInput<"draft">, signal?: AbortSignal): Promise<AtlasOutput<"draft">> =>
      response(() => sdk.client.atlas.draft({ ...input, directory: sdk.directory }, options(signal)), signal),
    emit: (input: AtlasInput<"emit">, signal?: AbortSignal): Promise<AtlasOutput<"emit">> =>
      response(() => sdk.client.atlas.emit({ ...input, directory: sdk.directory }, options(signal)), signal),
    link: (input: AtlasInput<"link">, signal?: AbortSignal): Promise<AtlasOutput<"link">> =>
      response(() => sdk.client.atlas.link({ ...input, directory: sdk.directory }, options(signal)), signal),
    verify: (input: AtlasInput<"verify">, signal?: AbortSignal): Promise<AtlasOutput<"verify">> =>
      response(() => sdk.client.atlas.verify({ ...input, directory: sdk.directory }, options(signal)), signal),
  }
}

```ts
import { createStore } from "solid-js/store"
import { pathKey } from "@/utils/path-key"
import { isAtlasEmpty, type AtlasTransport } from "./atlas-api"
import type { AtlasInput, AtlasOutput, AtlasSnapshot } from "./atlas-types"

export type AtlasOwner = { server: string; directory: string; transport: AtlasTransport }
export type AtlasWrite = "promote" | "draft" | "emit" | "link" | "verify"
export type AtlasController = ReturnType<typeof createAtlasController>
type State = {
  key?: string
  snapshot?: AtlasSnapshot
  // D5: the instance worktree has no .atlas/.
  empty: boolean
  // The server has no `atlas` routes.
  unavailable: boolean
  loading: boolean
  error?: string
  health?: AtlasOutput<"health">
  reasoning: Record<string, AtlasOutput<"reasoning">>
  own: Record<string, AtlasOutput<"own">>
  pending: AtlasWrite[]
}

// One selected owner (server + directory), like the Lean controller. D6: the snapshot is read on select, on
// refresh, and again after every write; reasoning, own briefings and health are read on demand and cached.
export function createAtlasController(describe: (error: unknown) => { message: string; unavailable?: boolean }) {
  const [state, setState] = createStore<State>({
    empty: false,
    unavailable: false,
    loading: false,
    reasoning: {},
    own: {},
    pending: [],
  })
  const requests = new Set<AbortController>()
  const life = { disposed: false, sequence: 0, accepted: 0 }
  let owner: AtlasOwner | undefined
  let read: AbortController | undefined

  const request = () => {
    const abort = new AbortController()
    requests.add(abort)
    return abort
  }
  const cancel = () => {
    requests.forEach((item) => item.abort())
    requests.clear()
  }
  const current = (active: AtlasOwner, abort: AbortController) =>
    !life.disposed && owner === active && !abort.signal.aborted
  const failure = (error: unknown) => {
    const result = describe(error)
    if (result.unavailable) {
      cancel()
      setState({ snapshot: undefined, empty: false, unavailable: true, loading: false, pending: [], error: undefined })
      return
    }
    setState("error", result.message)
  }
  // Runs one request against the selected owner; resolves undefined when it failed or was superseded.
  const run = async <T>(call: (transport: AtlasTransport, signal: AbortSignal) => Promise<T>, abort = request()) => {
    const active = owner
    if (!active || life.disposed || state.unavailable) return
    try {
      const value = await call(active.transport, abort.signal)
      if (current(active, abort)) return value
    } catch (error) {
      if (current(active, abort)) failure(error)
    } finally {
      requests.delete(abort)
    }
  }

  const refresh = async () => {
    if (!owner || life.disposed || state.unavailable) return
    read?.abort()
    const order = ++life.sequence
    setState({ loading: true, error: undefined })
    read = request()
    const data = await run((transport, signal) => transport.snapshot(signal), read)
    if (life.disposed) return
    // A newer refresh owns the loading flag until it finishes.
    if (order === life.sequence) setState("loading", false)
    if (data === undefined || order < life.accepted) return
    life.accepted = order
    if (isAtlasEmpty(data)) return setState({ snapshot: undefined, empty: true })
    setState({ snapshot: data, empty: false })
  }

  const reasoning = async (id: string) => {
    if (state.reasoning[id]) return state.reasoning[id]
    const data = await run((transport, signal) => transport.reasoning(id, signal))
    if (data !== undefined) setState("reasoning", id, data)
    return data
  }
  const own = async (scope: string) => {
    if (state.own[scope]) return state.own[scope]
    const data = await run((transport, signal) => transport.own(scope, signal))
    if (data !== undefined) setState("own", scope, data)
    return data
  }
  const health = async () => {
    const data = await run((transport, signal) => transport.health(signal))
    if (data !== undefined) setState("health", data)
    return data
  }

  // A write marks its kind pending, then rereads the snapshot (D6). `verify` saves nothing, so it does not reread.
  const write = async <T>(kind: AtlasWrite, call: (transport: AtlasTransport, signal: AbortSignal) => Promise<T>) => {
    setState({ pending: [...state.pending, kind], error: undefined })
    const data = await run(call)
    if (life.disposed) return data
    setState("pending", state.pending.filter((item) => item !== kind))
    if (kind !== "verify") await refresh()
    return data
  }

  return {
    state,
    refresh,
    reasoning,
    own,
    health,
    promote: (input: AtlasInput<"promote">) => write("promote", (transport, signal) => transport.promote(input, signal)),
    draft: (input: AtlasInput<"draft">) => write("draft", (transport, signal) => transport.draft(input, signal)),
    emit: (input: AtlasInput<"emit">) => write("emit", (transport, signal) => transport.emit(input, signal)),
    link: (input: AtlasInput<"link">) => write("link", (transport, signal) => transport.link(input, signal)),
    verify: (input: AtlasInput<"verify">) => write("verify", (transport, signal) => transport.verify(input, signal)),
    async select(next: AtlasOwner) {
      if (life.disposed) return
      cancel()
      owner = { ...next }
      life.accepted = 0
      setState({
        key: `${next.server}\0${pathKey(next.directory)}`,
        snapshot: undefined,
        empty: false,
        unavailable: false,
        loading: false,
        error: undefined,
        health: undefined,
        reasoning: {},
        own: {},
        pending: [],
      })
      await refresh()
    },
    dispose() {
      life.disposed = true
      owner = undefined
      cancel()
    },
  }
}
```

import { describe, expect, test } from "bun:test"
import { AtlasAPIError, isAtlasEmpty, type AtlasTransport } from "./atlas-api"
import { createAtlasController } from "./atlas-controller"
import { atlasFixture } from "./atlas-model.fixture"
import type { AtlasInput, AtlasOutput } from "./atlas-types"

// A plain transport object that records each call; tests replace single methods.
function fake(overrides: Partial<AtlasTransport> = {}) {
  const calls: string[] = []
  const answer = <T>(name: string, value: T) => {
    calls.push(name)
    return Promise.resolve(value)
  }
  const transport: AtlasTransport = {
    snapshot: () => answer("snapshot", atlasFixture()),
    reasoning: (id) => answer(`reasoning:${id}`, { id } as unknown as AtlasOutput<"reasoning">),
    own: (scope) => answer(`own:${scope}`, { scope } as unknown as AtlasOutput<"own">),
    health: () => answer("health", {} as unknown as AtlasOutput<"health">),
    promote: () => answer("promote", {} as unknown as AtlasOutput<"promote">),
    draft: () => answer("draft", {} as unknown as AtlasOutput<"draft">),
    emit: () => answer("emit", {} as unknown as AtlasOutput<"emit">),
    link: () => answer("link", {} as unknown as AtlasOutput<"link">),
    verify: () => answer("verify", {} as unknown as AtlasOutput<"verify">),
    ...overrides,
  }
  return { calls, transport }
}
const describeError = (error: unknown) =>
  error instanceof AtlasAPIError
    ? { message: "unavailable", unavailable: error.reason === "unsupported" }
    : { message: error instanceof Error ? error.message : "unknown" }
const owner = (transport: AtlasTransport) => ({ server: "local", directory: "/repo", transport })

describe("atlas controller", () => {
  test("loads the snapshot of the selected owner", async () => {
    const source = fake()
    const controller = createAtlasController(describeError)
    await controller.select(owner(source.transport))
    expect(controller.state.snapshot?.project).toBe("demo")
    expect([controller.state.empty, controller.state.unavailable, controller.state.loading]).toEqual([false, false, false])
    expect(source.calls).toEqual(["snapshot"])
  })

  test("an absent store reads as empty (D5)", async () => {
    const source = fake({ snapshot: () => Promise.resolve({ state: "empty" } as unknown as AtlasOutput<"snapshot">) })
    const controller = createAtlasController(describeError)
    await controller.select(owner(source.transport))
    expect(controller.state.empty).toBe(true)
    expect(controller.state.snapshot).toBeUndefined()
  })

  test("a server without the atlas routes is unavailable and is not asked again", async () => {
    const source = fake({ snapshot: () => Promise.reject(new AtlasAPIError("unsupported")) })
    const controller = createAtlasController(describeError)
    await controller.select(owner(source.transport))
    expect(controller.state.unavailable).toBe(true)
    await controller.refresh()
    await controller.health()
    expect(source.calls).toEqual([])
  })

  test("another failure is worded and kept", async () => {
    const source = fake({ snapshot: () => Promise.reject(new Error("boom")) })
    const controller = createAtlasController(describeError)
    await controller.select(owner(source.transport))
    expect([controller.state.error, controller.state.unavailable, controller.state.loading]).toEqual(["boom", false, false])
  })

  test("a write rereads the snapshot (D6); verify does not", async () => {
    const source = fake()
    const controller = createAtlasController(describeError)
    await controller.select(owner(source.transport))
    await controller.promote({} as AtlasInput<"promote">)
    expect(source.calls).toEqual(["snapshot", "promote", "snapshot"])
    expect(controller.state.pending).toEqual([])
    await controller.verify({} as AtlasInput<"verify">)
    expect(source.calls).toEqual(["snapshot", "promote", "snapshot", "verify"])
  })

  test("reasoning and own briefings are read once and cached", async () => {
    const source = fake()
    const controller = createAtlasController(describeError)
    await controller.select(owner(source.transport))
    await controller.reasoning("r1")
    await controller.reasoning("r1")
    await controller.own("packages/a")
    await controller.own("packages/a")
    expect(source.calls).toEqual(["snapshot", "reasoning:r1", "own:packages/a"])
    expect(Object.keys(controller.state.reasoning)).toEqual(["r1"])
  })

  test("a disposed controller asks nothing", async () => {
    const source = fake()
    const controller = createAtlasController(describeError)
    await controller.select(owner(source.transport))
    controller.dispose()
    await controller.refresh()
    expect(source.calls).toEqual(["snapshot"])
  })

  test("isAtlasEmpty only matches the empty answer", () => {
    expect(isAtlasEmpty({ state: "empty" })).toBe(true)
    expect([isAtlasEmpty(atlasFixture()), isAtlasEmpty(null), isAtlasEmpty("empty"), isAtlasEmpty({ state: "ready" })]).toEqual([false, false, false, false])
  })
})
