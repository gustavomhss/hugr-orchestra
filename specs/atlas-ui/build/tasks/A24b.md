## Porting rules applied
- Previous/next read the list order from the main pane's `[data-action="open"][data-value]` rows, the same list the frame's `j`/`k` keys use (the mock's `A.state.list`). The disabled state is read one frame after the fact changes, once the rows are on screen.
- D13: "Keep for later" and "Accept" on a staged candidate need R2; they keep their `cap` dot and toast the capability text instead of the mock's simulated write. "Rewrite" (R8) on a changed fact keeps its dot and opens the `edit` sheet, as in the mock's real mode.
- Retire: the mock refused a ratified or core fact with a toast before writing; that refusal stays here. The write itself needs A32b's `writeFact`, which lands after this task in lane order, so the other branch is a marked line (`// A24C: ...`) that toasts the frame's "not built yet" note; task A24c replaces it (`issues/a-u-views-1.md` item 1).
- Every action button is a `write` button: on a store in git it is disabled with the reason as its tooltip.
- The fact page's crumbs are the mock's `[Knowledge, <file name>]`; its "Back to knowledge" goes to `explore?view=list` (the mock's `#/knowledge`).

2. Create `packages/app-shell/src/orchestra/chapters/atlas/views/inspector.tsx` with exactly:
```tsx
import { createEffect, Match, on, Show, Switch } from "solid-js"
import { createStore } from "solid-js/store"
import { useLanguage } from "@/context/language"
import { useAtlas } from "../atlas-context"
import type { AtlasFact } from "../atlas-model"
import { atlasHref } from "../atlas-route"
import { type AtlasReq, Icon, useAtlasWords } from "../kit/chips"
import { Btn } from "../kit/controls"
import { openAtlasSheet } from "../sheets/state"
import { FactBody, useCopyLink } from "./fact-detail"
import type { AtlasInspectorProps } from "./index"

// The fact inspector beside a list (specs/atlas-ui/mock/views.js:210-219): step through the list, open the fact as a
// page, close; the actions follow the fact's state (staged, changed, replaced, or up to date).
export default function AtlasInspector(props: AtlasInspectorProps) {
  return (
    <Show when={props.model.byKey[props.factKey] as AtlasFact | undefined}>
      {(fact) => <Inspector {...props} fact={fact()} />}
    </Show>
  )
}

function Inspector(props: AtlasInspectorProps & { fact: AtlasFact }) {
  const atlas = useAtlas()
  const language = useLanguage()
  const words = useAtlasWords()
  const copyLink = useCopyLink()
  const [state, setState] = createStore({ order: [] as string[] })
  createEffect(on(() => props.factKey, () => requestAnimationFrame(() => setState("order", readOrder()))))
  const index = () => state.order.indexOf(props.factKey)
  const step = (delta: number) => {
    const order = readOrder()
    if (!order.length) return
    const at = order.indexOf(props.factKey)
    atlas.open(order[Math.max(0, Math.min(order.length - 1, at < 0 ? 0 : at + delta))])
  }
  const staged = () => props.fact.kind === "candidate" && !props.fact.promotedNow
  const factState = () => props.model.state(props.fact)
  // D13: a control that needs a new capability toasts what it needs instead of acting.
  const needs = (id: AtlasReq) =>
    atlas.toast(language.t("orchestra.atlas.kit.newcap.tip", { id, text: words.requirement(id) }), "sparkle")
  const retire = () => {
    if (props.model.governing(props.fact))
      return atlas.toast(
        language.t(
          props.fact.tier === "T0" ? "orchestra.atlas.fact.retire.refusedCore" : "orchestra.atlas.fact.retire.refusedRatified",
        ),
        "xc",
      )
    // A24C: the retire write goes here.
    atlas.toast(language.t("orchestra.atlas.frame.pending"), "info")
  }
  return (
    <>
      <header class="ins-head">
        <span class="ins-kind">
          <Icon name="fact" />
          <span>{props.model.unitOf(props.fact.file)}</span>
        </span>
        <Btn
          icon="up"
          ariaLabel={language.t("orchestra.atlas.fact.previous")}
          tip={language.t("orchestra.atlas.fact.previous")}
          disabled={index() <= 0}
          onClick={() => step(-1)}
        />
        <Btn
          icon="down"
          ariaLabel={language.t("orchestra.atlas.fact.next")}
          tip={language.t("orchestra.atlas.fact.next")}
          disabled={index() < 0 || index() >= state.order.length - 1}
          onClick={() => step(1)}
        />
        <Btn
          icon="expand"
          ariaLabel={language.t("orchestra.atlas.fact.page")}
          tip={language.t("orchestra.atlas.fact.page")}
          onClick={() => atlas.go(atlasHref("fact", props.fact.key))}
        />
        <Btn
          icon="close"
          ariaLabel={language.t("orchestra.atlas.fact.close")}
          tip={language.t("orchestra.atlas.fact.close")}
          onClick={() => atlas.close()}
        />
      </header>
      <div class="ins-body">
        <FactBody fact={props.fact} model={props.model} route={props.route} />
      </div>
      <footer class="ins-foot">
        <Btn
          icon="link"
          ariaLabel={language.t("orchestra.atlas.fact.copyLink")}
          tip={language.t("orchestra.atlas.fact.copyLink")}
          onClick={() => copyLink(props.fact.key)}
        />
        <span class="grow" />
        <Switch>
          <Match when={staged()}>
            <Btn label={language.t("orchestra.atlas.fact.keep")} write cap="R2" onClick={() => needs("R2")} />
            <Btn label={language.t("orchestra.atlas.fact.accept")} write primary cap="R2" onClick={() => needs("R2")} />
          </Match>
          <Match when={factState() === "drifted"}>
            <Btn
              label={language.t("orchestra.atlas.fact.rewrite")}
              write
              primary
              cap="R8"
              onClick={() => openAtlasSheet({ kind: "edit", key: props.fact.key })}
            />
          </Match>
          <Match when={factState() === "retired"}>
            <span class="at-hint">{language.t("orchestra.atlas.fact.replaced")}</span>
          </Match>
          <Match when={factState() === "fresh"}>
            <Btn label={language.t("orchestra.atlas.fact.edit")} write onClick={() => openAtlasSheet({ kind: "edit", key: props.fact.key })} />
            <Show when={!props.fact.slot}>
              <Btn label={language.t("orchestra.atlas.fact.classify")} write onClick={() => openAtlasSheet({ kind: "classify", key: props.fact.key })} />
            </Show>
            <Btn label={language.t("orchestra.atlas.fact.retire")} write onClick={retire} />
            <Show when={!props.model.governing(props.fact)}>
              <Btn
                label={language.t("orchestra.atlas.fact.ratify")}
                write
                primary
                onClick={() => openAtlasSheet({ kind: "ratify", key: props.fact.key })}
              />
            </Show>
          </Match>
        </Switch>
      </footer>
    </>
  )
}

// The fact keys of the rows in the main pane, in order, once each (the frame's j/k list).
function readOrder() {
  return [
    ...new Set(
      [...document.querySelectorAll<HTMLElement>('.atlas-main [data-action="open"][data-value]')].map((element) => element.dataset.value ?? ""),
    ),
  ]
}
```

3. Create `packages/app-shell/src/orchestra/chapters/atlas/views/fact.tsx` with exactly:
```tsx
import { createEffect, Show } from "solid-js"
import { useLanguage } from "@/context/language"
import { useAtlas } from "../atlas-context"
import type { AtlasFact } from "../atlas-model"
import { atlasHref } from "../atlas-route"
import { Btn, Empty } from "../kit/controls"
import { FactBody, useCopyLink } from "./fact-detail"
import type { AtlasViewProps } from "./index"

// A fact as a page (specs/atlas-ui/mock/views.js:222-229): the same body as the inspector, with Back and Copy link.
export default function AtlasFactView(props: AtlasViewProps) {
  const atlas = useAtlas()
  const language = useLanguage()
  const copyLink = useCopyLink()
  const fact = () => props.model.byKey[props.route.param] as AtlasFact | undefined
  createEffect(() => {
    const x = fact()
    atlas.crumbs(
      x
        ? [language.t("orchestra.atlas.fact.crumb.knowledge"), x.file.split("/").at(-1) ?? x.file]
        : [language.t("orchestra.atlas.fact.crumb.fact")],
    )
  })
  return (
    <Show
      when={fact()}
      fallback={
        <div class="at-page narrow">
          <Empty
            title={language.t("orchestra.atlas.fact.missing")}
            action={
              <Btn
                label={language.t("orchestra.atlas.fact.missing.back")}
                onClick={() => atlas.go(atlasHref("explore", "", { view: "list" }))}
              />
            }
          >
            {language.t("orchestra.atlas.fact.missing.text")}
          </Empty>
        </div>
      }
    >
      {(x) => (
        <div class="at-page narrow">
          <div class="mx-heading">
            <div>
              <div class="mx-eyebrow">{props.model.unitOf(x().file)}</div>
            </div>
            <div class="heading-actions">
              <Btn label={language.t("orchestra.atlas.fact.back")} icon="back" onClick={() => history.back()} />
              <Btn label={language.t("orchestra.atlas.fact.copyLink")} icon="link" onClick={() => copyLink(x().key)} />
            </div>
          </div>
          <div class="at-pagecard">
            <FactBody fact={x()} model={props.model} route={props.route} />
          </div>
        </div>
      )}
    </Show>
  )
}
```
