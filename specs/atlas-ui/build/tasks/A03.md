## Porting rules (applied in the code below)
- extract.py's two `rev-parse` per fact become ONE `git cat-file --batch-check=%(objectname)` for all facts; a line
  that is not a 40–64 hex object id means "no object" (extract.py's empty `rev-parse` output).
- A rev is used only when it matches `/^[0-9a-f]{7,64}$/`, and a file only when non-empty and free of newlines: a value
  read from the store never reaches git as an option.
- `git ls-files -z` replaces `git ls-files` (no quoted paths). Python `splitlines()` → split on `\r?\n`, last empty dropped.
- Hunks: at most 2, header lines (`---`, `+++`, `\`) skipped, at most 18 lines each, each cut to 160 characters.
- Commits: at most 3 `[%h, subject]` pairs from `git log --format=%h%x09%s <rev>..HEAD -- <file>`.
- `langs` keys drop the dot; `children` maps "." for the root.

```ts
// @atlas/adapter-io — src/surface/read-git.ts  (the Atlas UI surface: what git says about the store)
//
// The git half of specs/atlas-ui/mock/extract.py: HEAD, each fact's drift since the rev it was derived at, which store
// files git tracks, the tracked files per extension, and the children of every tracked directory. Every call goes
// through the one no-shell git seam (`../run-git.ts`).
//
// TOTAL: a failed git call reads as empty output, which is what extract.py read from a failed `subprocess.run`.
// Where extract.py ran two `rev-parse` per fact, this asks for both object ids of every fact in ONE
// `git cat-file --batch-check`; the diff, the shortstat and the log still run once per DRIFTED fact. A rev is used only
// when it is a hex object name, so a rev read from the store can never reach git as an option.

import { extname, posix } from "node:path"
import { runGit } from "../run-git.js"
import type { PendingFact } from "./read-store.js"
import type { AtlasDrift, AtlasDriftHunk, AtlasFact, AtlasTracked } from "./types.js"

const MAX_HUNKS = 2
const MAX_HUNK_LINES = 18
const MAX_LINE = 160
const MAX_COMMITS = 3
const LANG_EXTS = new Set([".ts", ".tsx", ".js", ".mjs", ".py", ".rs", ".go", ".sh"])
const REV = /^[0-9a-f]{7,64}$/
const OBJECT_ID = /^[0-9a-f]{40,64}$/
const PROJECTION_FILE = /^\.atlas\/projection(\.\d+)?\.json$/
const STAGING_FILE = /^\.atlas\/staging(\.\d+)?\.json$/

/** HEAD's first 8 characters; "" outside git. */
export function headShort(repoPath: string): string {
  return gitOut(repoPath, ["rev-parse", "HEAD"]).trim().slice(0, 8)
}

/** Every tracked path (`git ls-files -z`, so no path is quoted). */
export function trackedFiles(repoPath: string): readonly string[] {
  return nulSplit(gitOut(repoPath, ["ls-files", "-z"]))
}

/** Which durable store files git tracks (store-provenance.ts asks the same question for the read gate). */
export function trackedStore(repoPath: string): AtlasTracked {
  const files = nulSplit(gitOut(repoPath, ["ls-files", "-z", "--", ".atlas"]))
  return {
    projection: files.some((file) => PROJECTION_FILE.test(file)),
    cas: files.filter((file) => file.startsWith(".atlas/cas/")).length,
    staging: files.some((file) => STAGING_FILE.test(file)),
    policy: files.includes(".atlas/policy.json"),
    memory: files.includes(".atlas/memory.jsonl"),
  }
}

/** Tracked files per extension, for the extensions extract.py counts; the key drops the dot. */
export function langCounts(tracked: readonly string[]): Readonly<Record<string, number>> {
  return tracked
    .map((file) => extname(file))
    .filter((ext) => LANG_EXTS.has(ext))
    .reduce<Record<string, number>>((counts, ext) => ({ ...counts, [ext.slice(1)]: (counts[ext.slice(1)] ?? 0) + 1 }), {})
}

/** Directory ("." for the root) → the sorted names of its tracked children, files and directories alike. */
export function childrenOf(tracked: readonly string[]): Readonly<Record<string, readonly string[]>> {
  const kids = new Map<string, Set<string>>()
  for (const file of tracked) {
    const parts = file.split("/")
    parts.forEach((_, i) => {
      const prefix = parts.slice(0, i + 1).join("/")
      const parent = posix.dirname(prefix)
      kids.set(parent, (kids.get(parent) ?? new Set<string>()).add(posix.basename(prefix)))
    })
  }
  return Object.fromEntries([...kids].map(([dir, names]) => [dir, [...names].sort()]))
}

/** Each pending fact as the snapshot serves it: with `drift` when its file's blob at the rev differs from HEAD's. */
export function withDrift(repoPath: string, pending: readonly PendingFact[]): readonly AtlasFact[] {
  const asked = pending.filter((p) => REV.test(p.rev) && p.fact.file.length > 0 && !p.fact.file.includes("\n"))
  const ids =
    asked.length === 0
      ? []
      : gitOut(
          repoPath,
          ["cat-file", "--batch-check=%(objectname)"],
          `${asked.flatMap((p) => [`${p.rev}:${p.fact.file}`, `HEAD:${p.fact.file}`]).join("\n")}\n`,
        ).split("\n")
  const changed = new Set(
    asked.filter((_, i) => {
      const before = objectId(ids[2 * i])
      const after = objectId(ids[2 * i + 1])
      return before !== "" && after !== "" && before !== after
    }),
  )
  return pending.map((p) => (changed.has(p) ? { ...p.fact, drift: driftOf(repoPath, p.rev, p.fact.file) } : p.fact))
}

function driftOf(repoPath: string, rev: string, file: string): AtlasDrift {
  return {
    hunks: hunksOf(gitOut(repoPath, ["diff", "-U2", rev, "HEAD", "--", file])),
    stat: gitOut(repoPath, ["diff", "--shortstat", rev, "HEAD", "--", file]).trim(),
    commits: lines(gitOut(repoPath, ["log", "--format=%h%x09%s", `${rev}..HEAD`, "--", file]))
      .slice(0, MAX_COMMITS)
      .map(commitOf),
  }
}

/** extract.py's hunk loop: at most two hunks, header lines skipped, at most 18 lines each, each cut to 160. */
function hunksOf(diff: string): readonly AtlasDriftHunk[] {
  const hunks: { readonly at: string; readonly lines: string[] }[] = []
  for (const line of lines(diff)) {
    if (line.startsWith("@@")) {
      if (hunks.length === MAX_HUNKS) break
      hunks.push({ at: (line.split("@@")[1] ?? "").trim(), lines: [] })
      continue
    }
    const current = hunks.at(-1)
    if (current === undefined || line.startsWith("---") || line.startsWith("+++") || line.startsWith("\\")) continue
    if (current.lines.length < MAX_HUNK_LINES) current.lines.push(line.slice(0, MAX_LINE))
  }
  return hunks
}

function commitOf(line: string): readonly [string, string] {
  const tab = line.indexOf("\t")
  return tab < 0 ? [line, ""] : [line.slice(0, tab), line.slice(tab + 1)]
}

function objectId(line: string | undefined): string {
  const id = (line ?? "").trim()
  return OBJECT_ID.test(id) ? id : ""
}

/** Python's `splitlines()` for git output: split on line ends, no trailing empty line. */
function lines(text: string): readonly string[] {
  const all = text.split(/\r?\n/)
  return all.at(-1) === "" ? all.slice(0, -1) : all
}

function nulSplit(text: string): readonly string[] {
  return text.split("\0").filter((part) => part.length > 0)
}

/** One git call; any failure (not a repository, a bad rev, git absent) reads as "". */
function gitOut(repoPath: string, args: readonly string[], input?: string): string {
  try {
    return runGit(repoPath, args, input === undefined ? {} : { input })
  } catch {
    return ""
  }
}
```

```ts
// @atlas/adapter-io — test/surface-read-git.test.ts  (the Atlas UI surface: the git half of the snapshot)
//
// A real two-commit git repository: `src/a.ts` changes between R1 and R2, `src/b.ts` does not. The drift, the
// tracked-store flags, the language counts and the children tree are read back through `read-git.ts`.

import { execFileSync } from "node:child_process"
import { mkdirSync, mkdtempSync, rmSync, writeFileSync } from "node:fs"
import { tmpdir } from "node:os"
import { dirname, join } from "node:path"
import { afterEach, describe, expect, it } from "vitest"
import { childrenOf, headShort, langCounts, trackedFiles, trackedStore, withDrift } from "../src/surface/read-git.js"
import type { PendingFact } from "../src/surface/read-store.js"
import type { AtlasFact } from "../src/surface/types.js"

const dirs: string[] = []
afterEach(() => {
  for (const dir of dirs.splice(0)) rmSync(dir, { recursive: true, force: true })
})

function tempDir(): string {
  const dir = mkdtempSync(join(tmpdir(), "atlas-surface-git-"))
  dirs.push(dir)
  return dir
}

/** R1: src/a.ts, src/b.ts, README.md and three store files. R2: src/a.ts changed, committed as "change a". */
function twoCommits(): { dir: string; r1: string; r2: string; short2: string } {
  const dir = tempDir()
  const git = (...args: string[]): string =>
    execFileSync("git", args, { cwd: dir, encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] }).trim()
  const write = (rel: string, content: string): void => {
    mkdirSync(dirname(join(dir, rel)), { recursive: true })
    writeFileSync(join(dir, rel), content)
  }
  git("init", "-q")
  git("config", "user.email", "fix@atlas.test")
  git("config", "user.name", "atlas-fixture")
  git("config", "commit.gpgsign", "false")
  write("src/a.ts", "export const a = 1\n")
  write("src/b.ts", "export const b = 1\n")
  write("README.md", "# fixture\n")
  write(".atlas/projection.json", "{}\n")
  write(".atlas/policy.json", "{}\n")
  write(".atlas/cas/aa/aaaa", "{}\n")
  git("add", "-A")
  git("commit", "-q", "-m", "R1")
  const r1 = git("rev-parse", "HEAD")
  write("src/a.ts", "export const a = 2\n")
  git("commit", "-q", "-am", "change a")
  return { dir, r1, r2: git("rev-parse", "HEAD"), short2: git("log", "-1", "--format=%h") }
}

function pending(file: string, rev: string): PendingFact {
  const fact: AtlasFact = {
    key: `k-${file}-${rev}`,
    hash: "a".repeat(64),
    claim: "a claim",
    file,
    family: "advisory",
    tier: "T2",
    scope: "atlas:mined",
    derivedAt: rev.slice(0, 8),
    subtree: "",
    obvious: null,
    promoted: true,
    why: null,
    kind: null,
    authoring: null,
    storedFreshness: null,
    obviousBy: null,
    anchor: null,
    slot: null,
    seal: null,
    supersededBy: null,
    sameAs: null,
    provenance: null,
  }
  return { fact, rev }
}

describe("surface read-git — the git half of the snapshot", () => {
  it("adds drift only to a fact whose file changed since its rev, with extract.py's hunks, stat and commits", () => {
    const { dir, r1, r2, short2 } = twoCommits()
    const rows = [pending("src/a.ts", r1), pending("src/b.ts", r1), pending("src/a.ts", r2), pending("src/a.ts", "")]
    const out = withDrift(dir, rows)
    expect(out[0]?.drift).toEqual({
      hunks: [{ at: "-1 +1", lines: ["-export const a = 1", "+export const a = 2"] }],
      stat: "1 file changed, 1 insertion(+), 1 deletion(-)",
      commits: [[short2, "change a"]],
    })
    expect(out.slice(1).map((fact) => "drift" in fact)).toEqual([false, false, false])
    expect(out.map((fact) => fact.key)).toEqual(rows.map((row) => row.fact.key))
  })

  it("never passes a rev that is not a hex object name to git", () => {
    const { dir } = twoCommits()
    const out = withDrift(dir, [pending("src/a.ts", "--output=pwned"), pending("src/a.ts", "HEAD~1")])
    expect(out.map((fact) => "drift" in fact)).toEqual([false, false])
  })

  it("reads HEAD, the tracked files, the tracked store files, the language counts and the children tree", () => {
    const { dir, r2 } = twoCommits()
    expect(headShort(dir)).toBe(r2.slice(0, 8))
    const tracked = trackedFiles(dir)
    expect(tracked).toEqual([".atlas/cas/aa/aaaa", ".atlas/policy.json", ".atlas/projection.json", "README.md", "src/a.ts", "src/b.ts"])
    expect(trackedStore(dir)).toEqual({ projection: true, cas: 1, staging: false, policy: true, memory: false })
    expect(langCounts(tracked)).toEqual({ ts: 2 })
    expect(childrenOf(tracked)).toEqual({
      ".": [".atlas", "README.md", "src"],
      ".atlas": ["cas", "policy.json", "projection.json"],
      ".atlas/cas": ["aa"],
      ".atlas/cas/aa": ["aaaa"],
      src: ["a.ts", "b.ts"],
    })
  })

  it("counts extensions and builds children from paths alone", () => {
    expect(langCounts(["a.ts", "b.tsx", "c.d.ts", "d.md", ".sh", "e.sh", "f.mjs"])).toEqual({ ts: 2, tsx: 1, sh: 1, mjs: 1 })
    expect(childrenOf(["x/y/z.ts", "x/w.ts", "v"])).toEqual({ ".": ["v", "x"], x: ["w.ts", "y"], "x/y": ["z.ts"] })
  })

  it("reads a directory outside git as empty, and leaves every fact without drift", () => {
    const dir = tempDir()
    expect(headShort(dir)).toBe("")
    expect(trackedFiles(dir)).toEqual([])
    expect(trackedStore(dir)).toEqual({ projection: false, cas: 0, staging: false, policy: false, memory: false })
    const rev = "0123456789abcdef0123456789abcdef01234567"
    expect(withDrift(dir, [pending("src/a.ts", rev)]).map((fact) => "drift" in fact)).toEqual([false])
  })
})
```
