const runAt = <A>(name: string, call: (root: string) => A) =>
  Effect.gen(function* () {
    const root = yield* storeRoot
    return yield* Effect.try({
      try: () => call(root),
      catch: (cause) =>
        new Unavailable({
          message: (cause instanceof Error ? cause.message : String(cause)).split("\n")[0]!.slice(0, 300),
        }),
    })

const run = <A>(name: string, call: (surface: ReturnType<typeof openSurface>) => A) =>
  runAt(name, (root) => call(openSurface(root)))

import { existsSync } from "node:fs"
import { join } from "node:path"
import type { AtlasHealthResult } from "@orchestra/atlas-boundary/surface"
import type { Npm } from "@orchestra/core/npm"
import type { AppProcess } from "@orchestra/core/process"
import { Effect } from "effect"
import { ChildProcess } from "effect/unstable/process"
import type { Unavailable } from "@/server/atlas-surface"

// D38: Orchestra runs the indexer the store's plan names, from the repository root, so Genesis has the code graph it
// ranks sites by. TypeScript first, else Python, else nothing: both tools write the one `.atlas/index.scip` the readers
// open. The package comes from Orchestra's package cache at the version Atlas pins (adapter-io/src/scip.ts) and runs with
// Orchestra's own Bun, so the machine needs no Node. A failed run reports the tail of its stderr.

export type Result =
  | { readonly state: "empty" }
  | {
      readonly state: "ok"
      readonly ran: boolean
      readonly tool: string | null
      readonly exitCode: number | null
      /** Documents in the index after the run, or null when there is none. */
      readonly documents: number | null
      readonly message: string | null
    }

export type Services = {
  readonly npm: Npm.Interface
  readonly processes: AppProcess.Interface
  readonly health: () => Effect.Effect<AtlasHealthResult, Unavailable>
}

const PACKAGES = [
  ["scip-typescript", "@sourcegraph/scip-typescript"],
  ["scip-python", "@sourcegraph/scip-python"],
] as const

export const index = Effect.fn("AtlasIndex.index")(function* (services: Services, root: string) {
  const health = yield* services.health()
  if (health.state === "empty") return { state: "empty" as const }
  const planned = PACKAGES.flatMap(([tool, pkg]) =>
    health.index.configured.filter((lang) => lang.plan.tool === tool).map((lang) => ({ ...lang.plan, pkg })),
  )[0]
  if (!planned) return done(false, null, null, null, "no indexer is configured for this repository's languages")
  const bin = yield* services.npm.which(`${planned.pkg}@${planned.version}`, planned.tool)
  if (!bin) return done(false, planned.tool, null, null, `could not install ${planned.pkg}@${planned.version}`)
  const args = [
    ...planned.args,
    ...(planned.tool === "scip-typescript" && !existsSync(join(root, "tsconfig.json")) ? ["--infer-tsconfig"] : []),
  ]
  const ran = yield* services.processes
    .run(
      ChildProcess.make(process.execPath, [bin, ...args], {
        cwd: root,
        env: { BUN_BE_BUN: "1" },
        extendEnv: true,
        stdin: "ignore",
      }),
      { timeout: "10 minutes", maxOutputBytes: 1_000_000, maxErrorBytes: 64_000 },
    )
    .pipe(
      Effect.map((result) => ({ exitCode: result.exitCode, stderr: result.stderr.toString("utf8") })),
      Effect.catch((error) => Effect.succeed({ exitCode: -1, stderr: error.message })),
    )
  const after = yield* services.health()
  const documents = after.state === "ok" && after.index.scip.kind === "present" ? after.index.scip.documents : null
  const failure = ran.stderr.trim().slice(-300) || `${planned.tool} exited with ${ran.exitCode}`
  return done(true, planned.tool, ran.exitCode, documents, ran.exitCode === 0 ? null : failure)
})

function done(ran: boolean, tool: string | null, exitCode: number | null, documents: number | null, message: string | null) {
  return { state: "ok" as const, ran, tool, exitCode, documents, message }
}

export * as AtlasIndex from "./atlas-index"
```

import { ModelV2 } from "@orchestra/core/model"
import { ProviderV2 } from "@orchestra/core/provider"
import { LLMEvent } from "@orchestra/llm"
import { Effect, Stream } from "effect"
import type { Agent } from "@/agent/agent"
import { AtlasSurface } from "@/server/atlas-surface"
import type { Provider } from "@/provider/provider"
import type { LLM } from "@/session/llm"
import { MessageID, SessionID } from "@/session/schema"

// Genesis mining from the screen (DECISIONS.md D40). The boundary lists the sites the advisory pass visits with the
// exact prompt `atlas mine` would send; the host asks the chosen model once per site, four at a time, as plain text;
// the boundary runs the pass again answering each site from those replies, through Atlas's own answer gate and
// admission, and stages the candidates. A site whose call fails is left out and abstains. Nothing is promoted.

export type Input = { readonly providerID: string; readonly modelID: string; readonly ceiling: number }

export type Services = {
  readonly atlas: Pick<AtlasSurface.Interface, "mineSites" | "mineWith">
  readonly llm: LLM.Interface
  readonly provider: Provider.Interface
}

const MINE_AGENT: Agent.Info = {
  name: "atlas-genesis-mine",
  mode: "primary",
  permission: [],
  options: {},
  native: true,
  // An empty prompt would fall back to the model's general coding prompt.
  prompt:
    "You read one unit of a codebase and follow the message's instructions exactly. A program reads your reply, so reply only in the format the message asks for.",
}

export const mine = Effect.fn("AtlasMine.mine")(function* (services: Services, input: Input) {
  const ceiling = Math.min(200, Math.max(1, Math.floor(input.ceiling)))
  const listed = yield* services.atlas.mineSites(ceiling)
  if (listed.state === "empty") return listed
  const model = yield* services.provider
    .getModel(ProviderV2.ID.make(input.providerID), ModelV2.ID.make(input.modelID))
    .pipe(Effect.mapError((error) => new AtlasSurface.Unavailable({ message: firstLine(error) })))
  const replies = yield* Effect.forEach(
    listed.sites,
    (site) =>
      ask(services.llm, model, site.prompt).pipe(
        Effect.map((text) => [[site.key, text] as const]),
        Effect.catch(() => Effect.succeed([])),
      ),
    { concurrency: 4 },
  )
  return yield* services.atlas.mineWith({
    ceiling,
    answers: Object.fromEntries(replies.flat()),
    model: `${input.providerID}/${input.modelID}`,
  })
})

const ask = (llm: LLM.Interface, model: Provider.Model, prompt: string) => {
  const sessionID = SessionID.descending()
  return llm
    .stream({
      agent: MINE_AGENT,
      user: {
        id: MessageID.ascending(),
        sessionID,
        role: "user",
        time: { created: Date.now() },
        agent: MINE_AGENT.name,
        model: { providerID: model.providerID, modelID: model.id },
      },
      system: [],
      tools: {},
      model,
      sessionID,
      retries: 2,
      messages: [{ role: "user", content: prompt }],
    })
    .pipe(
      Stream.filter(LLMEvent.is.textDelta),
      Stream.map((event) => event.text),
      Stream.mkString,
    )
}

function firstLine(error: unknown) {
  const text = error instanceof Error && error.message ? error.message : String(error)
  return text.split("\n")[0]!.slice(0, 300)
}

export * as AtlasMine from "./atlas-mine"
```

// D40 mining without a real model or store: the surface lists two sites and records what the host hands back, the
// LLM service replays one fixed reply, and the provider is the test fake (`openai/gpt-5.2`).

const SITES = [
  { key: "src/a.ts::f", prompt: "prompt for f" },
  { key: "src/b.ts::g", prompt: "prompt for g" },
]
const MINED = { state: "ok" as const, sites: 2, facts: 1, complete: true, why: null, refusal: null }

async function mine(input: AtlasMine.Input, listed: "ok" | "empty" = "ok") {
  const handed: MineWithInput[] = []
  const prompts: string[] = []
  const ceilings: number[] = []
  const provider = await Effect.runPromise(
    Effect.gen(function* () {
      return yield* Provider.Service
    }).pipe(Effect.provide(ProviderTest.fake().layer)),
  )
  const services: AtlasMine.Services = {
    atlas: {
      mineSites: (ceiling) => {
        ceilings.push(ceiling)
        return Effect.succeed(listed === "ok" ? { state: "ok" as const, sites: SITES, promptDigest: "d" } : { state: "empty" as const })
      },
      mineWith: (input) => {
        handed.push(input)
        return Effect.succeed(MINED)
      },
    },
    llm: {
      stream: (request: LLM.StreamInput) => {
        prompts.push(JSON.stringify(request.messages))
        return Stream.make(LLMEvent.textDelta({ id: "t", text: "NO-FACT" }))
      },
    },
    provider,
  }
  const result = await Effect.runPromise(AtlasMine.mine(services, input))
  return { result, handed, prompts, ceilings }
}

describe("AtlasMine.mine", () => {
  test("asks the model once per site and hands every reply back keyed by site", async () => {
    const run = await mine({ providerID: "openai", modelID: "gpt-5.2", ceiling: 20 })
    expect(run.result).toEqual(MINED)
    expect(run.prompts).toHaveLength(2)
    expect(run.prompts.join(" ")).toContain("prompt for f")
    expect(run.prompts.join(" ")).toContain("prompt for g")
    expect(run.handed).toEqual([
      { ceiling: 20, answers: { "src/a.ts::f": "NO-FACT", "src/b.ts::g": "NO-FACT" }, model: "openai/gpt-5.2" },
    ])
  })

  test("keeps the ceiling between 1 and 200", async () => {
    expect((await mine({ providerID: "openai", modelID: "gpt-5.2", ceiling: 999 })).ceilings).toEqual([200])
    expect((await mine({ providerID: "openai", modelID: "gpt-5.2", ceiling: 0 })).ceilings).toEqual([1])
  })

  test("answers empty without calling the model when the project has no store", async () => {
    const run = await mine({ providerID: "openai", modelID: "gpt-5.2", ceiling: 20 }, "empty")
    expect(run.result).toEqual({ state: "empty" })
    expect(run.prompts).toEqual([])
    expect(run.handed).toEqual([])
  })
})
