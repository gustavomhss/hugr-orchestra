```tsx
import { For, type JSX, Show } from "solid-js"
import { Dynamic } from "solid-js/web"
import { useAtlas } from "../atlas-context"

// Copy strings keep the mock's inline markup: <code>, <b>, <em>, <mark> and <small> around plain text. Rich renders
// those five tags as elements and everything else as text, so neither copy nor a parameter can inject markup.
type RichTag = "code" | "b" | "em" | "mark" | "small"
export type RichPart = { tag?: RichTag; text: string }
const TAGS = /<(code|b|em|mark|small)>([\s\S]*?)<\/\1>/g

export function richParts(text: string): RichPart[] {
  const matches = [...text.matchAll(TAGS)]
  const starts = [0, ...matches.map((match) => (match.index ?? 0) + match[0].length)]
  return [
    ...matches.flatMap((match, index): RichPart[] => [
      { text: text.slice(starts[index], match.index ?? 0) },
      { tag: match[1] as RichTag, text: match[2] },
    ]),
    { text: text.slice(starts[matches.length]) },
  ].filter((part) => !!part.tag || part.text !== "")
}

export function Rich(props: { text: string }) {
  return (
    <For each={richParts(props.text)}>
      {(part) => (
        <Show when={part.tag} fallback={part.text}>
          {(tag) => <Dynamic component={tag()}>{part.text}</Dynamic>}
        </Show>
      )}
    </For>
  )
}

// A link to another Atlas page (the mock's href="#/..."): a real href for open-in-new-tab, navigation through the
// Atlas context on a plain click.
export function AtlasLink(props: { href: string; class?: string; replace?: boolean; children: JSX.Element }) {
  const atlas = useAtlas()
  return (
    <a
      class={props.class}
      href={props.href}
      onClick={(event) => {
        if (event.metaKey || event.ctrlKey || event.shiftKey || event.button !== 0) return
        event.preventDefault()
        atlas.go(props.href, props.replace)
      }}
    >
      {props.children}
    </a>
  )
}
```

import { describe, expect, test } from "bun:test"
import { richParts } from "./text"

describe("atlas rich copy", () => {
  test("plain text stays one part", () => {
    expect(richParts("Nothing changed")).toEqual([{ text: "Nothing changed" }])
    expect(richParts("")).toEqual([])
  })

  test("the five tags become parts, in order", () => {
    expect(richParts("Keep <code>.atlas/</code> out of git")).toEqual([
      { text: "Keep " },
      { tag: "code", text: ".atlas/" },
      { text: " out of git" },
    ])
    expect(richParts("<b>a</b><code>b</code>")).toEqual([
      { tag: "b", text: "a" },
      { tag: "code", text: "b" },
    ])
    expect(richParts("x <em>e</em> <mark>m</mark> <small>s</small>")).toEqual([
      { text: "x " },
      { tag: "em", text: "e" },
      { text: " " },
      { tag: "mark", text: "m" },
      { text: " " },
      { tag: "small", text: "s" },
    ])
  })

  test("other tags and broken pairs stay text", () => {
    expect(richParts("atlas node <address> and <i>x</i>")).toEqual([{ text: "atlas node <address> and <i>x</i>" }])
    expect(richParts("<code>a</b>")).toEqual([{ text: "<code>a</b>" }])
    expect(richParts('<img src=x onerror="alert(1)">')).toEqual([{ text: '<img src=x onerror="alert(1)">' }])
  })
})
```

import { createStore } from "solid-js/store"
import type { AtlasMemberId } from "../atlas-model"

// Every sheet (dialog) a view can open, by kind, with what it needs. One sheet shows at a time for the whole
// chapter, like the mock's single #dialog; AtlasSheetHost (host.tsx) renders it.
export type AtlasSheet =
  | { kind: "ratify"; key: string }
  | { kind: "edit"; key: string }
  | { kind: "classify"; key: string }
  | { kind: "link"; key: string }
  | { kind: "add-fact"; path: string }
  | { kind: "verify"; scope: string }
  | { kind: "promote-all" }
  | { kind: "promote-result" }
  | { kind: "reconcile" }
  | { kind: "mine" }
  | { kind: "producers" }
  | { kind: "hotset" }
  | { kind: "republish" }
  | { kind: "respawn"; member: AtlasMemberId; unit: string }
  | { kind: "rule"; member: AtlasMemberId; lesson: string; unit: string }
export type AtlasSheetKind = AtlasSheet["kind"]
export type AtlasSheetProps<K extends AtlasSheetKind> = { sheet: Extract<AtlasSheet, { kind: K }>; onClose: () => void }
export type AtlasPromoteResult = Exclude<AtlasOutput<"promote">, { state: "empty" }>
// The last `atlas promote` run (the mock's A.state.promote): Review shows its counts and each row's outcome.
// `staged` holds the keys that were staged candidates before the run.
export type AtlasPromoteRecord = { run: AtlasPromoteResult; staged: string[] }

const [state, setState] = createStore<{ sheet?: AtlasSheet; promote?: AtlasPromoteRecord }>({})

export const atlasSheetState = state
export const openAtlasSheet = (sheet: AtlasSheet) => setState("sheet", sheet)
export const closeAtlasSheet = () => setState("sheet", undefined)
export const keepPromoteRun = (record: AtlasPromoteRecord) => setState("promote", record)
export const resetAtlasSheets = () => setState({ sheet: undefined, promote: undefined })
```

4. Create `packages/app-shell/src/orchestra/chapters/atlas/sheets/host.tsx` with exactly:
```tsx
import { type Component, onCleanup, Show } from "solid-js"
import { Dynamic } from "solid-js/web"
import { type AtlasSheet, type AtlasSheetKind, type AtlasSheetProps, atlasSheetState, closeAtlasSheet, resetAtlasSheets } from "./state"

// Each sheet task adds its own line here: `<kind>: <Component>,` with the component imported from its file.
// A kind without a line opens nothing.
export const atlasSheets: { [K in AtlasSheetKind]?: Component<AtlasSheetProps<K>> } = {}

// Mounted once by the chapter frame. `keyed` remounts the dialog when one sheet replaces another, so each opens
// with showModal and its own fresh state.
export function AtlasSheetHost() {
  onCleanup(() => resetAtlasSheets())
  return (
    <Show when={atlasSheetState.sheet} keyed>
      {(sheet) => (
        <Show when={atlasSheets[sheet.kind] as Component<{ sheet: AtlasSheet; onClose: () => void }> | undefined}>
          {(view) => <Dynamic component={view()} sheet={sheet} onClose={closeAtlasSheet} />}
        </Show>
      )}
    </Show>
  )
}
```
