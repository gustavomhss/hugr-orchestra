```ts
import type { AtlasEvent, AtlasModel } from "../atlas-model"
import { seatName } from "../kit/format"

// Activity rows: the agents' events (session receipts, empty until they are served, D12) and one row per commit
// that moved anchored code (mock views-more.js:65-69).
export type ActivityRow =
  | { id: string; kind: "code"; sha: string; subject: string; facts: number }
  | { id: string; kind: AtlasEvent["kind"]; event: AtlasEvent }
export type ActivityShow = "all" | "reads" | "memory" | "holds" | "code"
export const ACTIVITY_SHOWS = ["all", "reads", "memory", "holds", "code"] as const

const KINDS = { reads: ["read"], memory: ["memory", "resume"], holds: ["hold"], code: ["code"] } as const

export function activityRows(model: AtlasModel): ActivityRow[] {
  return [
    ...model.activity().map((event, index): ActivityRow => ({ id: `e${index}`, kind: event.kind, event })),
    ...model.codeChanges().map(
      (change, index): ActivityRow => ({ id: `c${index}`, kind: "code", sha: change.sha, subject: change.subject, facts: change.facts.length }),
    ),
  ]
}

// "All" shows every agent event but not the code rows, which have their own filter.
export function shownRows(rows: ActivityRow[], show: ActivityShow) {
  if (show === "all") return rows.filter((row) => row.kind !== "code")
  return rows.filter((row) => (KINDS[show] as readonly string[]).includes(row.kind))
}

export const countRows = (rows: ActivityRow[], show: Exclude<ActivityShow, "all">) => shownRows(rows, show).length

// An event's sentence, as a copy key and its parameters (mock model.js:310-322). The view translates it.
export function eventText(event: AtlasEvent) {
  const name = seatName(event.who)
  if (event.kind === "memory" && event.recall) return { key: "orchestra.atlas.activity.event.recalled", params: { name } } as const
  if (event.kind === "memory" && event.ok) return { key: "orchestra.atlas.activity.event.saved", params: { name } } as const
  if (event.kind === "memory") return { key: "orchestra.atlas.activity.event.notSaved", params: { name } } as const
  if (event.kind === "resume" && event.ok) return { key: "orchestra.atlas.activity.event.resumed", params: { name } } as const
  if (event.kind === "resume") return { key: "orchestra.atlas.activity.event.noCheckpoint", params: { name } } as const
  if (event.kind === "hold") return { key: "orchestra.atlas.activity.event.held", params: { name } } as const
  if (event.action === "anchor")
    return {
      key: "orchestra.atlas.activity.event.froze",
      params: { name, count: event.units?.length ?? 0, target: event.target ?? "" },
    } as const
  if (event.action === "catalog") return { key: "orchestra.atlas.activity.event.catalog", params: { name } } as const
  return { key: "orchestra.atlas.activity.event.briefing", params: { name, unit: event.target ?? "" } } as const
}

// The mock's time(ms): "Oct 11 · 14:05 UTC", always in UTC.
export function eventTime(at: number) {
  const date = new Date(at)
  return {
    day: date.toLocaleDateString("en-US", { month: "short", day: "numeric", timeZone: "UTC" }),
    clock: date.toISOString().slice(11, 16),
  }
}
```

```ts
import { describe, expect, test } from "bun:test"
import { type AtlasEvent, createAtlasModel } from "../atlas-model"
import { atlasFixture } from "../atlas-model.fixture"
import { activityRows, countRows, eventText, eventTime, shownRows } from "./activity-data"

const event = (patch: Partial<AtlasEvent>): AtlasEvent => ({
  at: 0,
  who: "backend",
  kind: "memory",
  ok: true,
  unit: "T-1",
  ...patch,
})

describe("atlas activity data", () => {
  test("adds one code row per commit and keeps code rows out of All", () => {
    const rows = activityRows(createAtlasModel(atlasFixture()))
    expect(rows).toEqual([{ id: "c0", kind: "code", sha: "c0ffee1", subject: "fix x", facts: 1 }])
    expect(shownRows(rows, "all")).toEqual([])
    expect(shownRows(rows, "code").map((row) => row.id)).toEqual(["c0"])
    expect((["reads", "memory", "holds", "code"] as const).map((show) => countRows(rows, show))).toEqual([0, 0, 0, 1])
  })

  test("filters agent events by kind", () => {
    const rows = [
      { id: "e0", kind: "read" as const, event: event({ kind: "read", action: "catalog" }) },
      { id: "e1", kind: "resume" as const, event: event({ kind: "resume" }) },
      { id: "e2", kind: "hold" as const, event: event({ kind: "hold", ok: false }) },
    ]
    expect(shownRows(rows, "all").map((row) => row.id)).toEqual(["e0", "e1", "e2"])
    expect(shownRows(rows, "memory").map((row) => row.id)).toEqual(["e1"])
    expect(shownRows(rows, "reads").map((row) => row.id)).toEqual(["e0"])
    expect(shownRows(rows, "holds").map((row) => row.id)).toEqual(["e2"])
  })

  test("words each event kind with the seat's name", () => {
    expect(eventText(event({}))).toEqual({ key: "orchestra.atlas.activity.event.saved", params: { name: "Backend" } })
    expect(eventText(event({ ok: false })).key).toBe("orchestra.atlas.activity.event.notSaved")
    expect(eventText(event({ recall: true })).key).toBe("orchestra.atlas.activity.event.recalled")
    expect(eventText(event({ kind: "resume", who: "jimmy" }))).toEqual({
      key: "orchestra.atlas.activity.event.resumed",
      params: { name: "Jimmy" },
    })
    expect(eventText(event({ kind: "resume", ok: false })).key).toBe("orchestra.atlas.activity.event.noCheckpoint")
    expect(eventText(event({ kind: "hold", ok: false })).key).toBe("orchestra.atlas.activity.event.held")
    expect(eventText(event({ kind: "read", who: "archie", action: "anchor", target: "WP-1", units: ["a", "b"] }))).toEqual({
      key: "orchestra.atlas.activity.event.froze",
      params: { name: "Archie", count: 2, target: "WP-1" },
    })
    expect(eventText(event({ kind: "read", action: "catalog" })).key).toBe("orchestra.atlas.activity.event.catalog")
    expect(eventText(event({ kind: "read", action: "unit", target: "packages/a" }))).toEqual({
      key: "orchestra.atlas.activity.event.briefing",
      params: { name: "Backend", unit: "packages/a" },
    })
  })

  test("formats times in UTC", () => {
    expect(eventTime(Date.UTC(2026, 9, 11, 14, 5))).toEqual({ day: "Oct 11", clock: "14:05" })
  })
})
```

```tsx
import { createEffect, For, Show } from "solid-js"
import { useLanguage } from "@/context/language"
import { useAtlas } from "../atlas-context"
import type { AtlasEvent } from "../atlas-model"
import { atlasHref } from "../atlas-route"
import { Help, Icon, useAtlasWords } from "../kit/chips"
import { Seven } from "../kit/controls"
import { num, seatName } from "../kit/format"
import { AtlasLink } from "../kit/text"
import { ACTIVITY_SHOWS, type ActivityRow, activityRows, countRows, eventText, eventTime, shownRows } from "./activity-data"
import type { AtlasViewProps } from "./index"

// Activity: what Atlas gave agents and took from them (specs/atlas-ui/mock/views-more.js:58-83). Agent events are
// session receipts, which are not served yet (D12); code rows come from the store.
const LABELS = {
  all: "orchestra.atlas.activity.show.all",
  reads: "orchestra.atlas.activity.show.reads",
  memory: "orchestra.atlas.activity.show.memory",
  holds: "orchestra.atlas.activity.show.holds",
  code: "orchestra.atlas.activity.show.code",
} as const

export default function AtlasActivityView(props: AtlasViewProps) {
  const atlas = useAtlas()
  const language = useLanguage()
  createEffect(() => atlas.crumbs([language.t("orchestra.atlas.activity.crumb")]))
  const show = () => ACTIVITY_SHOWS.find((item) => item === props.route.query.show) ?? "all"
  const rows = () => activityRows(props.model)
  const list = () => shownRows(rows(), show())
  return (
    <div class="at-page narrow">
      <div class="mx-heading">
        <div>
          <div class="mx-eyebrow">{language.t("orchestra.atlas.activity.eyebrow")}</div>
          <h1>{language.t("orchestra.atlas.activity.title")}</h1>
          <p>{language.t("orchestra.atlas.activity.line")}</p>
        </div>
      </div>
      <div class="at-toolbar">
        <div class="wf-range">
          <For each={ACTIVITY_SHOWS}>
            {(item) => (
              <button
                type="button"
                aria-pressed={show() === item}
                onClick={() => atlas.go(atlasHref("activity", "", { show: item }))}
              >
                {language.t(LABELS[item])}
                <Show when={item !== "all" && item}>
                  {(filter) => <span class="wf-count">{num(countRows(rows(), filter()))}</span>}
                </Show>
              </button>
            )}
          </For>
        </div>
        <Help text={language.t("orchestra.atlas.activity.help")} />
      </div>
      <section class="at-panel">
        <Show when={list().length} fallback={<div class="at-band-empty">{language.t("orchestra.atlas.activity.empty")}</div>}>
          <Seven id={`act${show()}`} rows={list()} first={25}>
            {(row) => (row.kind === "code" ? <CodeRow row={row} /> : <EventRow event={row.event} />)}
          </Seven>
        </Show>
      </section>
    </div>
  )
}

function CodeRow(props: { row: Extract<ActivityRow, { kind: "code" }> }) {
  const language = useLanguage()
  return (
    <AtlasLink class="at-event" href={atlasHref("review", "", { tab: "changed", commit: props.row.sha })}>
      <span class="at-avatar sm">
        <Icon name="commit" />
      </span>
      <div class="at-event-main">
        <span>{language.plural("orchestra.atlas.activity.code", props.row.facts, { sha: props.row.sha, n: num(props.row.facts) })}</span>
        <small>{props.row.subject}</small>
      </div>
      <Icon name="arrow" />
    </AtlasLink>
  )
}

function EventRow(props: { event: AtlasEvent }) {
  const language = useLanguage()
  const words = useAtlasWords()
  const text = () => eventText(props.event)
  return (
    <div class="at-event">
      <span class="at-avatar sm">{seatName(props.event.who)[0]}</span>
      <div class="at-event-main">
        <span>
          {language.t(text().key, text().params)}
          <Show when={props.event.unit && props.event.kind !== "read"}>
            {" "}
            <small>· {props.event.unit}</small>
          </Show>
        </span>
        <small>{language.t("orchestra.atlas.activity.time", eventTime(props.event.at))}</small>
      </div>
      <Show when={props.event.kind === "memory" && !props.event.ok}>
        <span class="at-outcome bad">{words.refusal(props.event.refusal ?? "")}</span>
      </Show>
      <Show when={props.event.kind === "hold"}>
        <span class="at-outcome">{language.t("orchestra.atlas.activity.onHold")}</span>
      </Show>
    </div>
  )
}
```
