## Mapping (what the code below does)
| Model member | From the snapshot |
|---|---|
| `snapshot.revision` | first 8 characters of `own.sourceRevision`; `""` when `own` is null or HOLD |
| `snapshot.units` | the unit ids of `own.units`, in file order, only when `own.status === "READY"` |
| `snapshot.catalog` | `territories.length > 0` |
| `snapshot.packs[id]` | `summary` = `pack.unit`, `owner` = `pack.shape.owner`, `tier` = `pack.shape.tier`, `invariants` = `pack.invariants` (`nodeId`, `tier`, `claim`, `freshness`), `advisory` = `pack.advisory.length`, `dependents` = `pack.edges.dependents.length`, `tokenEstimate` = `pack.tokenEstimate`, `sourceFiles` = number of `sourceBlobs` keys. A unit without a string `unit.id` is skipped. |
| `memory[m].rules` | records of member `m` with kind `project`: `rule` = `text`, `scope` = `scope ?? "*"`, `age` = how many records in the WHOLE log (every member) have a larger `seq`, `grounding` = `refs[0]`, `from` = `text` |
| `memory[m].tasks` | kind `task`: `taskId`, `attempted`, `failedWith`, `stoppedAt` from `entry`; `lesson` = `text`; `ref` = `refs[0] ?? ""`; `source` = `".atlas/memory.jsonl"` |
| `memory[m].prs` | kind `pr`: `prId`, `decisions`, `reviewOutcomes` from `entry`; `title` = `""` and `knowledgeDelta` = `[]` (the log holds neither a title nor a per-unit count) |
| `memory.maestro.logbook` | `text` of the member `orch`'s `logbook` records |
| `memory[m].bound` | unchanged: `true` except `maestro` |
| `territories()` | `territories`, with a tier other than `T0`/`T1` read as `T2` |

Not derived: `activity()` stays `[]`. Orientation events carry a channel and a label, no member, time or outcome, and memory records carry no time (only logbook entries do), so neither yields an `AtlasEvent`. The `orientation` key is not read by the model.

```ts

type AtlasMemoryWire = AtlasSnapshot["memory"][string][number]

// OWN-SNAPSHOT.json units ({ unit, sourceBlobs, pack: OwnPackPlus }) arrive as plain JSON (`unknown` on the wire),
// so each field is read defensively; a unit with no string id is skipped.
function ownPacks(units: readonly unknown[]): Record<string, AtlasOwnPack> {
  return Object.fromEntries(
    units.flatMap((entry) => {
      const unit = record(record(entry).unit)
      const pack = record(record(entry).pack)
      const shape = record(pack.shape)
      if (typeof unit.id !== "string") return []
      const own: AtlasOwnPack = {
        summary: text(pack.unit),
        owner: text(shape.owner),
        tier: tierOf(shape.tier),
        invariants: list(pack.invariants).map((row) => {
          const item = record(row)
          return { nodeId: text(item.nodeId), tier: tierOf(item.tier), claim: text(item.claim), freshness: text(item.freshness) }
        }),
        advisory: list(pack.advisory).length,
        dependents: list(record(pack.edges).dependents).length,
        tokenEstimate: typeof pack.tokenEstimate === "number" ? pack.tokenEstimate : 0,
        sourceFiles: Object.keys(record(record(entry).sourceBlobs)).length,
      }
      return [[unit.id, own] as const]
    }),
  )
}

function taskOf(r: AtlasMemoryWire): AtlasMemoryTask {
  const entry = record(r.entry)
  return {
    taskId: text(entry.taskId),
    attempted: strings(entry.attempted),
    failedWith: strings(entry.failedWith),
    stoppedAt: text(entry.stoppedAt),
    lesson: r.text,
    ref: r.refs[0] ?? "",
    source: ".atlas/memory.jsonl",
  }
}

// A PR record has no title, and its knowledge delta is a list of facts, not per-unit counts: both stay empty.
function prOf(r: AtlasMemoryWire): AtlasMemoryPr {
  const entry = record(r.entry)
  return {
    prId: text(entry.prId),
    title: "",
    decisions: strings(entry.decisions),
    reviewOutcomes: strings(entry.reviewOutcomes),
    knowledgeDelta: [],
  }
}

function record(value: unknown): Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value) ? (value as Record<string, unknown>) : {}
}

function list(value: unknown): unknown[] {
  return Array.isArray(value) ? value : []
}

function strings(value: unknown) {
  return list(value).filter((item): item is string => typeof item === "string")
}

function text(value: unknown) {
  return typeof value === "string" ? value : ""
}

function tierOf(value: unknown): "T0" | "T1" | "T2" {
  return value === "T0" || value === "T1" ? value : "T2"
}
```
