```ts
import { isAtlasEmpty } from "../atlas-api"
import type { AtlasController } from "../atlas-controller"

// One fact write through the surface (D2): `draft` composes the fact at the anchor, `emit` writes it through the
// governed door, which runs every gate again. `atlas draft` always drafts T2 with a placeholder group
// (tools/src/draft.ts); a write at another tier, or into the fact's own group, sets them on the drafted fact before
// emit, as a CLI author does (issue item 6). The draft's own dry run is T2 only, so the emit verdict decides.
export type FactWrite = {
  anchor: string
  slot: string
  claim: string
  token: string
  supersede?: boolean
  tier?: string
  scope?: string
}
export type FactWriteOutcome = { ok: true } | { ok: false; reason: string }

export function governed(fact: unknown, patch: { tier?: string; scope?: string }): unknown {
  if (typeof fact !== "object" || fact === null) return fact
  return { ...fact, ...(patch.tier ? { tier: patch.tier } : {}), ...(patch.scope ? { scope: patch.scope } : {}) }
}

// Resolves undefined when a request failed or the store is gone: the controller has already set what the frame shows.
export async function writeFact(controller: AtlasController, input: FactWrite): Promise<FactWriteOutcome | undefined> {
  const drafted = await controller.draft({
    anchor: input.anchor,
    slot: input.slot,
    claim: input.claim,
    supersede: input.supersede,
    token: input.token,
  })
  if (!drafted || isAtlasEmpty(drafted)) return
  if (!drafted.ok || !drafted.data) return { ok: false, reason: drafted.rejected ?? "unknown" }
  const emitted = await controller.emit({
    fact: governed(drafted.data.draft.fact, { tier: input.tier, scope: input.scope }),
    at: drafted.data.draft.rev,
    token: input.token,
  })
  if (!emitted || isAtlasEmpty(emitted)) return
  if (!emitted.ok || !emitted.data?.emitted) return { ok: false, reason: emitted.data?.rejected ?? emitted.rejected ?? "unknown" }
  return { ok: true }
}
```

```ts
import { describe, expect, test } from "bun:test"
import type { AtlasTransport } from "../atlas-api"
import { createAtlasController } from "../atlas-controller"
import { atlasFixture } from "../atlas-model.fixture"
import type { AtlasOutput } from "../atlas-types"
import { governed, writeFact } from "./write"

const guidance = { next: "", invariant: "" }
const drafted = {
  state: "ok",
  ok: true,
  data: {
    draft: { fact: { kind: "advisory", tier: "T2", scope: "README.md" }, rev: "rev1", operation: "UPDATE", route: "full-ratify" },
    check: null,
  },
  guidance,
} as unknown as AtlasOutput<"draft">
const emittedOk = { state: "ok", ok: true, data: { emitted: true, nodeKey: "k4" }, guidance } as unknown as AtlasOutput<"emit">

// A transport that answers the two writes and records what they were sent.
function fake(draft: AtlasOutput<"draft">, emit: AtlasOutput<"emit">) {
  const sent: { name: string; input: unknown }[] = []
  const unused = <T>() => Promise.resolve({} as unknown as T)
  const transport: AtlasTransport = {
    snapshot: () => Promise.resolve(atlasFixture()),
    reasoning: () => unused(),
    own: () => unused(),
    health: () => unused(),
    promote: () => unused(),
    link: () => unused(),
    verify: () => unused(),
    draft: (input) => {
      sent.push({ name: "draft", input })
      return Promise.resolve(draft)
    },
    emit: (input) => {
      sent.push({ name: "emit", input })
      return Promise.resolve(emit)
    },
  }
  return { sent, transport }
}
async function controllerOf(transport: AtlasTransport) {
  const controller = createAtlasController((error) => ({ message: error instanceof Error ? error.message : "unknown" }))
  await controller.select({ server: "local", directory: "/repo", transport })
  return controller
}

describe("atlas fact write", () => {
  test("drafts, then emits the drafted fact at the target tier and group", async () => {
    const source = fake(drafted, emittedOk)
    const outcome = await writeFact(await controllerOf(source.transport), {
      anchor: "README.md",
      slot: "",
      claim: "The readme explains setup.",
      token: "k",
      tier: "T1",
      scope: "atlas:mined",
    })
    expect(outcome).toEqual({ ok: true })
    expect(source.sent).toEqual([
      { name: "draft", input: { anchor: "README.md", slot: "", claim: "The readme explains setup.", token: "k" } },
      { name: "emit", input: { fact: { kind: "advisory", tier: "T1", scope: "atlas:mined" }, at: "rev1", token: "k" } },
    ])
  })

  test("stops at a refused draft", async () => {
    const refused = { state: "ok", ok: false, rejected: "ungrounded", guidance } as unknown as AtlasOutput<"draft">
    const source = fake(refused, emittedOk)
    const outcome = await writeFact(await controllerOf(source.transport), { anchor: "a", slot: "gotcha", claim: "c", token: "" })
    expect(outcome).toEqual({ ok: false, reason: "ungrounded" })
    expect(source.sent.map((item) => item.name)).toEqual(["draft"])
  })

  test("reports the emit door's refusal", async () => {
    const emitRefused = { state: "ok", ok: true, data: { emitted: false, rejected: "unratified" }, guidance } as unknown as AtlasOutput<"emit">
    const outcome = await writeFact(await controllerOf(fake(drafted, emitRefused).transport), { anchor: "a", slot: "", claim: "c", token: "" })
    expect(outcome).toEqual({ ok: false, reason: "unratified" })
  })

  test("an empty store resolves nothing", async () => {
    const empty = { state: "empty" } as unknown as AtlasOutput<"draft">
    expect(await writeFact(await controllerOf(fake(empty, emittedOk).transport), { anchor: "a", slot: "", claim: "c", token: "" })).toBeUndefined()
  })

  test("patches only objects, and only the fields given", () => {
    expect(governed(null, { tier: "T1" })).toBeNull()
    expect(governed({ tier: "T2", a: 1 }, {})).toEqual({ tier: "T2", a: 1 })
    expect(governed({ tier: "T2", scope: "x" }, { tier: "T0" })).toEqual({ tier: "T0", scope: "x" })
  })
})
```

```tsx
import { For, type JSX, onMount, Show } from "solid-js"
import { useLanguage } from "@/context/language"
import { useAtlas } from "../atlas-context"
import type { AtlasChecks, AtlasFact, AtlasModel } from "../atlas-model"
import { Icon, useAtlasWords } from "../kit/chips"

// The parts the write sheets share (specs/atlas-ui/mock/app.js:174-181).
const GATE_WORDS = {
  shape: "orchestra.atlas.sheets.gate.shape",
  truth: "orchestra.atlas.sheets.gate.truth",
  authz: "orchestra.atlas.sheets.gate.authz",
  ratify: "orchestra.atlas.sheets.gate.ratify",
} as const

// Renders a sheet's body with the model and the fact it is about; nothing once the fact is gone.
export function WithFact(props: {
  factKey: string
  children: (model: () => AtlasModel, fact: () => AtlasFact) => JSX.Element
}) {
  const atlas = useAtlas()
  return (
    <Show when={atlas.model()}>
      {(model) => <Show when={model().byKey[props.factKey]}>{(fact) => props.children(model, fact)}</Show>}
    </Show>
  )
}

// The Checks strip: `atlas check` in four words, recomputed as the key is typed. It previews the gates from the
// model; the door's own verdict comes back when the write is sent.
export function ChecksStrip(props: { checks: AtlasChecks }) {
  const language = useLanguage()
  const words = useAtlasWords()
  return (
    <div class="at-checks-strip">
      <For each={props.checks.gates}>
        {(gate) => (
          <span
            class={`at-check ${gate.reached ? (gate.pass ? "good" : "bad") : "idle"}`}
            data-tip={gate.pass ? gate.gate : `${gate.gate} · ${gate.reason ?? ""} · ${words.remedy(gate)}`}
          >
            <Icon name={gate.reached ? (gate.pass ? "checkc" : "xc") : "minusc"} />
            {language.t(GATE_WORDS[gate.gate])}
          </span>
        )}
      </For>
    </div>
  )
}

// Who the write is judged as, and the approval key (the ratifier name for core) for this one write. The actor is
// the store's own (`health().actor`: ATLAS_ACTOR, else the git e-mail), read once when the sheet opens.
export function WritingAs(props: { core?: boolean; token: string; onToken: (value: string) => void }) {
  const atlas = useAtlas()
  const language = useLanguage()
  onMount(() => {
    if (!atlas.controller.state.health) void atlas.controller.health()
  })
  const actor = () => {
    const health = atlas.controller.state.health
    return health && "actor" in health ? health.actor : (atlas.model()?.actor ?? "")
  }
  return (
    <div class="mx-fields">
      <label class="mx-field">
        <span>{language.t("orchestra.atlas.sheets.writingAs")}</span>
        <input value={actor()} disabled />
      </label>
      <label class="mx-field">
        <span>{language.t(props.core ? "orchestra.atlas.sheets.ratifier" : "orchestra.atlas.sheets.approvalKey")}</span>
        <input
          type="password"
          autocomplete="off"
          value={props.token}
          placeholder={language.t(
            props.core ? "orchestra.atlas.sheets.ratifier.placeholder" : "orchestra.atlas.sheets.approvalKey.placeholder",
          )}
          onInput={(event) => props.onToken(event.currentTarget.value)}
        />
      </label>
    </div>
  )
}

// A store in git refuses every write (untrusted-store); the sheet says so before anything is sent.
export function LockedNote() {
  const atlas = useAtlas()
  const language = useLanguage()
  const words = useAtlasWords()
  return (
    <Show when={atlas.model() && !atlas.model()?.writable()}>
      <div class="ins-note">
        <Icon name="lock" />
        <span>{language.t("orchestra.atlas.sheets.locked", { reason: words.refusal("untrusted-store") })}</span>
      </div>
    </Show>
  )
}

// The footer of a sheet that only reports: optional actions, then Done.
export function DoneFoot(props: { onDone: () => void; children?: JSX.Element }) {
  const language = useLanguage()
  return (
    <footer class="mx-dialog-foot">
      <span class="grow" />
      {props.children}
      <button type="button" class="mx-btn primary" onClick={() => props.onDone()}>
        {language.t("orchestra.atlas.sheets.done")}
      </button>
    </footer>
  )
}

// Copies a command and says so (the mock's copy action).
export function useCopy() {
  const atlas = useAtlas()
  const language = useLanguage()
  return (text: string) => {
    const done = () => atlas.toast(language.t("orchestra.atlas.sheets.copied"), "copy")
    navigator.clipboard.writeText(text).then(done, done)
  }
}
```
