   // @atlas/adapter-io — src/open-questions-source.ts  (R11: the store feed for `deriveOpenQuestions`, and the interview)
   //
   // Reads what the pure question rules (`@atlas/genesis` open-questions.ts) need from one repository: the live current
   // facts (retired ones excluded, R6), each unit's policy owner (`terrainOwner`, the owner `atlas own` prints), and each
   // file's blast — how many files depend on it in the code index's reverse closure. It builds its own index from the
   // working tree and `.atlas/index.scip` (no dump ⇒ no edges ⇒ every blast is 0), so one call costs one index build;
   // it is not a leg of the composed runtime because `compose.ts` has no room under the godfile ceiling.
   //
   // `interviewOf` (D29) serves one batch over those questions for the CLI and the Atlas screen alike. A question is
   // answered only when its LATEST `answer` record has no `drafted`: a model's draft keeps it open and rides along as
   // `draft`, so a person still answers it (an accepted draft is a plain answer of the same text).

   import { join } from "node:path"
   import type { Hash } from "@atlas/contracts"
   import { assembleInterview, deriveOpenQuestions } from "@atlas/genesis"
   import type { QuestionFact, RankedQuestion } from "@atlas/genesis"
   import { build, createDepgraph, nodeHashOfPath } from "@atlas/index"
   import { currentNodes, isRetired } from "@atlas/knowledge"
   import type { GroundedFact } from "@atlas/knowledge"
   import { walkFileTree } from "./fs.js"
   import { CAS_REL } from "./okf-store.js"
   import { terrainOwner } from "./own-source.js"
   import { loadPolicy } from "./policy.js"
   import { foldReview, readReview } from "./review-log.js"
   import type { ReviewFold } from "./review-log.js"
   import { headSha } from "./run-git.js"
   import { readScipOrEmpty, SCIP_INDEX_REL } from "./scip.js"
   import { createDiskStore, rehydrateProjection } from "./store.js"
   import { gitSidecarTrust } from "./store-provenance.js"

   /** One interview question as the CLI and the Atlas screen serve it: a model's latest draft rides along. */
   export type InterviewQuestion = RankedQuestion & {
     readonly draft?: {
       readonly answer: string
       readonly model: string
       readonly why: string
       readonly by: string
       readonly at: string
     }
   }

   /** The open questions for the repository at `repoPath`, unanswered ones only, highest rank first. */
   export function openQuestionsOf(repoPath: string, answered: ReadonlySet<string>): readonly RankedQuestion[] {
     const store = createDiskStore(join(repoPath, CAS_REL), () => headSha(repoPath), gitSidecarTrust(repoPath))
     const facts = currentNodes(rehydrateProjection(store)).flatMap((node): QuestionFact[] => {
       const fact = store.get(node.contentHash as Hash) as GroundedFact | undefined
       if (node.primaryAnchor === undefined || fact === undefined || isRetired(fact)) return []
       return [
         {
           nodeKey: node.nodeKey,
           file: node.primaryAnchor.split("::")[0] ?? node.primaryAnchor,
           ...(node.slot !== undefined ? { slot: node.slot } : {}),
           tier: node.tier ?? fact.tier,
           claim: node.claims[0] ?? "",
         },
       ]
     })
     const policy = loadPolicy(repoPath)
     const depgraph = createDepgraph(build(walkFileTree(repoPath), readScipOrEmpty(join(repoPath, SCIP_INDEX_REL))).edges)
     return deriveOpenQuestions({
       facts,
       ownerOf: (unit) => terrainOwner(policy, unit),
       blast: (file) => {
         const self = nodeHashOfPath(file)
         return depgraph.reverseClosure(self).closure.filter((h) => h !== self).length
       },
       answered,
     })
   }

   /**
    * One interview batch (`assembleInterview`, at most `INTERVIEW_CAP`) over the open questions, plus how many wait for
    * the next session. A log with lines that do not verify folds what did verify; the writes over it refuse instead.
    * `questions` defaults to the store read above, and exists so a caller's own tests can supply the questions.
    */
   export function interviewOf(
     repoPath: string,
     questions: (answered: ReadonlySet<string>) => readonly RankedQuestion[] = (answered) => openQuestionsOf(repoPath, answered),
   ): { readonly batch: readonly InterviewQuestion[]; readonly deferred: number } {
     const fold = foldReview(readReview(repoPath).records)
     const assembled = assembleInterview(questions(answeredIds(fold)))
     // `assembleInterview` slices the ranked list it is given, so the batch holds the same RankedQuestion objects.
     const batch = (assembled.batch.questions as readonly RankedQuestion[]).map((q): InterviewQuestion => {
       const latest = fold.answers.get(q.id)
       const drafted = latest?.drafted
       if (latest === undefined || drafted === undefined) return q
       return { ...q, draft: { answer: latest.answer, model: drafted.model, why: drafted.why, by: latest.by, at: latest.at } }
     })
     return { batch, deferred: assembled.deferred.length }
   }

   /** The question ids a PERSON has answered: a question whose latest answer is a model's draft is still open (D29). */
   export function answeredIds(fold: ReviewFold): ReadonlySet<string> {
     return new Set([...fold.answers.values()].filter((a) => a.drafted === undefined).map((a) => a.questionId))
   }

   // @atlas/adapter-io — src/review-answer.ts  (R11 / D29: answer one open interview question)
   //
   // The answer half of `atlas answer`, moved out of the CLI (lead decision D29) so the CLI runner and the Atlas
   // screen's surface answer through ONE function. An answer is one `answer` record in the review log. A PERSON's
   // `tier` answer of T1 or T0 also appends a `proposal`, so a rule still needs `atlas decide` by a person, and reuses
   // an open proposal for the same node and tier instead of adding a second one. A model's DRAFTED answer only appends
   // its `answer` record with `drafted`: it never proposes, and the question stays open until a person answers it.

   import type { RankedQuestion } from "@atlas/genesis"
   import { answeredIds, openQuestionsOf } from "./open-questions-source.js"
   import { appendReview, foldReviewLog, resolveReviewId } from "./review-log.js"
   import type { ReviewOf, ReviewRefusal } from "./review-log.js"

   /** One answer. `questionId` is an exact id or a unique prefix of at least 8 characters. */
   export interface AnswerInput {
     readonly questionId: string
     readonly answer: string
     readonly by: string
     readonly at: string
     readonly drafted?: { readonly model: string; readonly why: string }
   }

   export type AnswerResult =
     | {
         readonly ok: true
         readonly question: RankedQuestion
         readonly answer: ReviewOf<"answer">
         /** The proposal a person's T1/T0 tier answer appended. */
         readonly proposal?: ReviewOf<"proposal">
         /** The proposal already open for that node and tier, which the answer reuses. */
         readonly pending?: ReviewOf<"proposal">
       }
     | ReviewRefusal

   /** Answer the open question `input.questionId`. `questions` defaults to the store's open questions, and exists so a
    *  caller's own tests can supply them. */
   export function answerQuestion(
     repoPath: string,
     input: AnswerInput,
     questions: (answered: ReadonlySet<string>) => readonly RankedQuestion[] = (answered) => openQuestionsOf(repoPath, answered),
   ): AnswerResult {
     const answer = input.answer.trim()
     if (answer.length === 0) return invalid("answer requires the answer text after the question id")
     const drafted = input.drafted
     if (drafted !== undefined && (drafted.model.trim() === "" || drafted.why.trim() === "")) {
       return invalid("answer: a drafted answer names its model and why")
     }
     const read = foldReviewLog(repoPath)
     if ("refusal" in read) return read
     const open = questions(answeredIds(read.fold))
     const id = resolveReviewId(
       open.map((q) => q.id),
       input.questionId,
     )
     const question = open.find((q) => q.id === id)
     if (question === undefined) {
       return invalid(
         `answer: no open question matches '${input.questionId}' — \`atlas interview\` lists the open batch (an answered question drops out)`,
       )
     }
     // A draft never proposes: only a person's answer starts a ratification (D29).
     const toTier = drafted === undefined && question.kind === "tier" && (answer === "T1" || answer === "T0") ? answer : undefined
     const nodeKey = question.nodeKey
     const pending =
       toTier === undefined || nodeKey === undefined
         ? undefined
         : read.fold.openProposals.find((p) => p.nodeKey === nodeKey && p.toTier === toTier)
     const proposal =
       toTier === undefined || nodeKey === undefined || pending !== undefined
         ? undefined
         : (appendReview(repoPath, {
             kind: "proposal",
             nodeKey,
             toTier,
             reason: `ratification interview answer to question ${question.id}`,
             at: input.at,
             by: input.by,
           }) as ReviewOf<"proposal">)
     const record = appendReview(repoPath, {
       kind: "answer",
       questionId: question.id,
       answer,
       ...(drafted !== undefined ? { drafted: { model: drafted.model, why: drafted.why } } : {}),
       at: input.at,
       by: input.by,
     }) as ReviewOf<"answer">
     return {
       ok: true,
       question,
       answer: record,
       ...(proposal !== undefined ? { proposal } : {}),
       ...(pending !== undefined ? { pending } : {}),
     }
   }

   function invalid(reason: string): ReviewRefusal {
     return { ok: false, refusal: "invalid", reason }
   }
   ```

   // @atlas/cli — src/interview-cli.ts  (R11: `atlas interview [--json]` / `atlas answer <questionId> <answer...> --by <actor>`)
   //
   // The ratification interview (D17): the open questions derived from the store (`openQuestionsOf`), served as ONE batch
   // of at most `INTERVIEW_CAP` by `interviewOf` (adapter-io, D29), ranked by blast × tier; the rest waits for the next
   // session. An answer goes through `answerQuestion` (adapter-io, D29), the one function the Atlas screen answers
   // through too: one `answer` record in the review log, and an answered question drops out of the next batch. A `tier`
   // question answered `T1` or `T0` ALSO appends a `proposal`, so a rule still needs `atlas decide` by a person — the
   // interview itself never ratifies anything. A model's draft (written by the Atlas screen's delegation) is printed
   // under its question, which stays open until a person answers it.

   import { answerQuestion, interviewOf, openQuestionsOf } from "@atlas/adapter-io"
   import { INTERVIEW_CAP } from "@atlas/genesis"
   import type { RankedQuestion } from "@atlas/genesis"
   import type { CliVerdict } from "./render.js"
   import { foldOrRefuse, governedRefusal, requireBy, reviewOutcome, usageError } from "./review-common.js"

   /** What interview/answer read and write through — injected so a test needs no store or index. */
   export interface InterviewPorts {
     readonly repoPath: string
     readonly now: () => string
     /** The open questions, minus the answered ids, highest rank first. */
     readonly questions: (answered: ReadonlySet<string>) => readonly RankedQuestion[]
   }

   /** The production ports: the repository at cwd, the wall clock, the real store/policy/index feed. */
   export function interviewPorts(): InterviewPorts {
     const repoPath = process.cwd()
     return { repoPath, now: () => new Date().toISOString(), questions: (answered) => openQuestionsOf(repoPath, answered) }
   }

   export function runInterview(ports: InterviewPorts, flags: Readonly<Record<string, string>>): CliVerdict {
     const read = foldOrRefuse(ports.repoPath)
     if ("refusal" in read) return read.refusal
     const interview = interviewOf(ports.repoPath, ports.questions)
     const batch = interview.batch
     if (flags["json"] === "true") {
       return { exitCode: 0, stdout: `${JSON.stringify({ batch, deferred: interview.deferred })}\n` }
     }
     return reviewOutcome(
       batch.length === 0
         ? "no open questions — every unit has an owner and a rationale, and no rule candidate or contested slot is waiting"
         : `${batch.length} question(s) in this session's batch (at most ${INTERVIEW_CAP}); ${interview.deferred} deferred to the next session — answer with \`atlas answer <questionId> <answer...> --by <you>\``,
       batch.flatMap((q) => [
         `  ${q.id} ${q.kind} rank ${q.rank}: ${q.ask}`,
         ...(q.options !== undefined ? [`      options: ${q.options.join(" | ")}`] : []),
         `      why: ${q.rankReason}`,
         ...(q.draft !== undefined ? [`      draft by ${q.draft.model}: ${q.draft.answer} (${q.draft.why})`] : []),
       ]),
     )
   }

   export function runAnswer(ports: InterviewPorts, positionals: readonly string[], flags: Readonly<Record<string, string>>): CliVerdict {
     const by = requireBy(flags)
     if (by === undefined) return usageError("answer requires --by <actor>: the person who answers")
     const wanted = positionals[0] ?? ""
     const answer = positionals.slice(1).join(" ").trim()
     if (answer.length === 0) return usageError("answer requires the answer text after the question id")
     const out = answerQuestion(ports.repoPath, { questionId: wanted, answer, by, at: ports.now() }, ports.questions)
     if (!out.ok) return out.refusal === "invalid" ? usageError(out.reason) : governedRefusal(out.reason)
     return reviewOutcome(
       out.proposal !== undefined
         ? `answered; proposal ${out.proposal.id} now waits for \`atlas decide\` by a person`
         : "answered; the question drops out of the next batch",
       [
         `answer: ${out.answer.id} ${out.question.kind} ${out.question.subject} by ${by}: ${answer}`,
         ...(out.proposal !== undefined ? [`proposal: ${out.proposal.id} ${out.proposal.nodeKey} -> ${out.proposal.toTier}`] : []),
         ...(out.pending !== undefined ? [`proposal: ${out.pending.id} already open for ${out.pending.nodeKey} -> ${out.pending.toTier}`] : []),
       ],
     )
   }
   ```
