```ts
import type { AtlasFact, AtlasModel } from "../atlas-model"
import type { AtlasOutput } from "../atlas-types"

// The five stops of one fact's life (mock views-explore.js:453-464): [icon, title copy key].
export const HOW_STOPS = [
  ["sparkle", "orchestra.atlas.how.stop.learned"],
  ["shield", "orchestra.atlas.how.stop.checked"],
  ["stack", "orchestra.atlas.how.stop.stored"],
  ["agents", "orchestra.atlas.how.stop.served"],
  ["drift", "orchestra.atlas.how.stop.watched"],
] as const

// The fact to follow: the one the query names, else the first that drifted and kept its reasoning, else the first.
export function followFact(model: AtlasModel, key: string | undefined): AtlasFact | undefined {
  const named: AtlasFact | undefined = key ? model.byKey[key] : undefined
  return named ?? model.live().find((x) => x.drift && x.why) ?? model.live()[0]
}

// "Another fact": a stable jump through the live facts, from the current key (mock views-explore.js:471).
export function anotherFact(model: AtlasModel, fact: AtlasFact) {
  const live = model.live()
  return live[Math.floor(((fact.key.charCodeAt(3) || 0) * 7) % live.length)]
}

export const howStep = (value: string | undefined) =>
  Math.min(HOW_STOPS.length - 1, Math.max(0, Math.floor(Number(value ?? 0)) || 0))

// Whether the unit's briefing sends this sentence in a paid band (ratified, watch out or hints).
export function servedIn(model: AtlasModel, fact: AtlasFact) {
  const briefing = model.briefing(model.unitOf(fact.file))
  return [...briefing.inv, ...briefing.gotchas, ...briefing.adv].includes(fact)
}

export const blastOf = (model: AtlasModel, fact: AtlasFact) => model.R.dependents[fact.file]?.length ?? 0

export const reasoningText = (value: AtlasOutput<"reasoning"> | undefined) =>
  value?.state === "ok" ? value.text : undefined
```

```ts
import { describe, expect, test } from "bun:test"
import { createAtlasModel } from "../atlas-model"
import { atlasFixture } from "../atlas-model.fixture"
import { anotherFact, blastOf, followFact, howStep, reasoningText, servedIn } from "./how-data"

describe("atlas follow one fact", () => {
  test("follows the named fact, else the first live fact", () => {
    const model = createAtlasModel(atlasFixture())
    expect(followFact(model, "k3")?.key).toBe("k3")
    expect(followFact(model, "nope")?.key).toBe("k1")
    expect(followFact(model, undefined)?.key).toBe("k1")
    expect(anotherFact(model, model.byKey.k2).key).toBe("k1")
  })

  test("clamps the stop to the five stops", () => {
    expect([howStep(undefined), howStep("2"), howStep("9"), howStep("x"), howStep("-1")]).toEqual([0, 2, 4, 0, 0])
  })

  test("says whether the briefing sends the fact and how much depends on its file", () => {
    const model = createAtlasModel(atlasFixture())
    expect(["k1", "k2", "k3", "k4"].map((key) => servedIn(model, model.byKey[key]))).toEqual([true, true, true, false])
    expect([blastOf(model, model.byKey.k1), blastOf(model, model.byKey.k4)]).toEqual([1, 0])
  })

  test("reads reasoning text only from a found entry", () => {
    expect(reasoningText({ state: "ok", id: "r1", text: "because" })).toBe("because")
    expect(reasoningText({ state: "missing", id: "r1" })).toBeUndefined()
    expect(reasoningText(undefined)).toBeUndefined()
  })
})
```

```tsx
export default function AtlasHowView(props: AtlasViewProps) {
  const atlas = useAtlas()
  const language = useLanguage()
  createEffect(() => atlas.crumbs([language.t("orchestra.atlas.how.crumb")]))
  return (
    <Show
      when={followFact(props.model, props.route.query.pick)}
      fallback={
        <div class="at-page wide">
          <p class="mx-empty">{language.t("orchestra.atlas.how.none")}</p>
        </div>
      }
    >
      {(fact) => <Follow model={props.model} fact={fact()} step={howStep(props.route.query.step)} />}
    </Show>
  )
}

function Follow(props: { model: AtlasModel; fact: AtlasFact; step: number }) {
  const atlas = useAtlas()
  const language = useLanguage()
  const href = (step: number) => atlasHref("how", "", { pick: props.fact.key, step: String(step) })
  const stop = () => HOW_STOPS[props.step]
  return (
    <div class="at-page wide">
      <section class="at-gen-hero">
        <div>
          <div class="mx-eyebrow">{language.t("orchestra.atlas.how.eyebrow")}</div>
          <h1>{language.t("orchestra.atlas.how.title")}</h1>

            label={language.t("orchestra.atlas.how.another")}

        <code class="at-mono">{props.fact.file}</code>
      </div>
      <ol class="at-how-track">
        <For each={HOW_STOPS}>

              <AtlasLink class={index() === props.step ? "on" : index() < props.step ? "done" : ""} href={href(index())}>

          <Show
            when={props.step < HOW_STOPS.length - 1}

            />
          </Show>
        </div>
      </section>
    </div>
  )
}

function Learned(props: { fact: AtlasFact }) {
  const atlas = useAtlas()
  const language = useLanguage()
  // D3: reasoning is read one entry at a time and cached by the controller.
  const [reasoning] = createResource(
    () => props.fact.why ?? undefined,
    (id) => atlas.controller.reasoning(id),
  )
  const text = () => reasoningText(reasoning()) ?? language.t("orchestra.atlas.how.learned.none")
  return (
    <>
      <p class="at-lead">{language.t("orchestra.atlas.how.learned.line")}</p>
      <div class="at-how-card">
        <p class="at-gen-claim">{language.t("orchestra.atlas.how.quote", { claim: props.fact.claim })}</p>
        <div class="at-reasoning short">{text().length > 520 ? `${text().slice(0, 520)}…` : text()}</div>
        <small>{language.t("orchestra.atlas.how.learned.answer")}</small>
      </div>
    </>
  )
}
```
